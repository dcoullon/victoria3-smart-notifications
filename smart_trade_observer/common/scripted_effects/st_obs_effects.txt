# Smart Treaties observer: one debug_log line per goods transfer article.
# Called with a goods transfer ARTICLE as scope. Read by
# tools/treaty_obs_report.py. Fields are ';'-separated inside s= and t=.
#
#   ST_OBS|<KIND>|<date>|<sender>|<receiver>|<goods key>|start=|q=|lane=|bind=|left=|inscope=|hist=
#     |s=price;buy;sell;imports;exports;production;consumption   (sender's market)
#     |t=price;buy;sell;imports;exports;production;consumption   (receiver's market)
#     |mm=<merchant marine price in the sender's market>
#     |lanemm=<the lane's merchant marine demand>                 (lane articles only)
#
# KIND: SIGN (treaty entered into force), MON (monthly), END (article
# removed or treaty dissolved).
#
# Learned on the first run (2026-11-10, game 1.15):
# - SCOPE.sC('x') / SCOPE.sGoods('x') on a scope saved with save_scope_as
#   inside this effect printed EMPTY, with no error in error.log. Names now
#   come straight off the article: THIS.GetTreatyArticle.<promote>.
# - THIS.ScriptValue and THIS.GetTreatyArticle.MakeScope.ScriptValue both
#   print; large numbers print plain (123456.78), no separator, no K.
# - Country.GetTag does not exist in this table; Treaty.GetStartDate is a
#   function returning text, so it ends the chain (no .GetString).
# - A treaty cannot hold a variable ("This scope doesn't support variables.
#   Scope: Treaty"), so the negative-treaty alert keys on a list on the
#   player's country instead.
# - GetShippingLane.GetMerchantMarineDemandRaw resolved, so it is logged for
#   lane articles: the real route cost behind the draft's x1.25 guess.
st_obs_log_article = {
	if = {
		limit = { exists = shipping_lane }
		debug_log = "ST_OBS|$KIND$|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetTreatyArticle.GetSourceCountry.GetNameNoFormatting]|[THIS.GetTreatyArticle.GetTargetCountry.GetNameNoFormatting]|[THIS.GetTreatyArticle.GetGoods.GetKey]|start=[THIS.GetTreatyArticle.GetTreaty.GetStartDate]|q=[THIS.ScriptValue('st_obs_q')|2]|lane=1|bind=[THIS.ScriptValue('st_obs_bind')|0]|left=[THIS.ScriptValue('st_obs_bind_left')|0]|inscope=[THIS.ScriptValue('st_obs_inscope')|0]|hist=[THIS.ScriptValue('st_obs_hist')|0]|s=[THIS.ScriptValue('st_obs_s_price')|2];[THIS.ScriptValue('st_obs_s_buy')|2];[THIS.ScriptValue('st_obs_s_sell')|2];[THIS.ScriptValue('st_obs_s_imp')|2];[THIS.ScriptValue('st_obs_s_exp')|2];[THIS.ScriptValue('st_obs_s_prod')|2];[THIS.ScriptValue('st_obs_s_cons')|2]|t=[THIS.ScriptValue('st_obs_t_price')|2];[THIS.ScriptValue('st_obs_t_buy')|2];[THIS.ScriptValue('st_obs_t_sell')|2];[THIS.ScriptValue('st_obs_t_imp')|2];[THIS.ScriptValue('st_obs_t_exp')|2];[THIS.ScriptValue('st_obs_t_prod')|2];[THIS.ScriptValue('st_obs_t_cons')|2]|mm=[THIS.ScriptValue('st_obs_s_mm')|2]|lanemm=[THIS.GetTreatyArticle.GetShippingLane.GetMerchantMarineDemandRaw|2]"
	}
	else = {
		debug_log = "ST_OBS|$KIND$|[TimeKeeper.GetCurrentDate.GetString]|[THIS.GetTreatyArticle.GetSourceCountry.GetNameNoFormatting]|[THIS.GetTreatyArticle.GetTargetCountry.GetNameNoFormatting]|[THIS.GetTreatyArticle.GetGoods.GetKey]|start=[THIS.GetTreatyArticle.GetTreaty.GetStartDate]|q=[THIS.ScriptValue('st_obs_q')|2]|lane=0|bind=[THIS.ScriptValue('st_obs_bind')|0]|left=[THIS.ScriptValue('st_obs_bind_left')|0]|inscope=[THIS.ScriptValue('st_obs_inscope')|0]|hist=[THIS.ScriptValue('st_obs_hist')|0]|s=[THIS.ScriptValue('st_obs_s_price')|2];[THIS.ScriptValue('st_obs_s_buy')|2];[THIS.ScriptValue('st_obs_s_sell')|2];[THIS.ScriptValue('st_obs_s_imp')|2];[THIS.ScriptValue('st_obs_s_exp')|2];[THIS.ScriptValue('st_obs_s_prod')|2];[THIS.ScriptValue('st_obs_s_cons')|2]|t=[THIS.ScriptValue('st_obs_t_price')|2];[THIS.ScriptValue('st_obs_t_buy')|2];[THIS.ScriptValue('st_obs_t_sell')|2];[THIS.ScriptValue('st_obs_t_imp')|2];[THIS.ScriptValue('st_obs_t_exp')|2];[THIS.ScriptValue('st_obs_t_prod')|2];[THIS.ScriptValue('st_obs_t_cons')|2]|mm=[THIS.ScriptValue('st_obs_s_mm')|2]"
	}
}
