# TEMPORARY diagnostic probes -- DELETE once the third-party notification
# filtering is scoped (Phase 4, see TODO.md). Adds no player-visible
# behaviour: every block below only writes debug_log lines.
#
# Purpose: the third-party ("something happened between two other
# countries") notifications are the main remaining filtering target, but
# several of them are rooted on object types we have no confirmed way to
# reach the involved countries from -- `# Root = Diplomatic Action` and
# `# Root = Diplomatic Pact` per vanilla's own comments in
# common/on_actions/00_code_on_actions.txt. The script_docs event_targets
# dump documents no scope links out of a diplomatic_action at all, so the
# only way to learn what is actually bound is to observe a real one.
#
# We already have a confirmed error from guessing here (`is_player trigger
# [ Wrong scope for trigger: diplomatic_action, expected country ]`), so
# every probe below is written defensively:
#   - `scope:x ?= { ... }` -- the `?=` guard means a scope that does not
#     exist is skipped rather than erroring (confirmed vanilla pattern,
#     e.g. `scope:target ?= { ... }` in vanilla's own on_war_end).
#   - the name is read as `[THIS.GetCountry.GetNameNoFormatting]` from INSIDE that
#     scope, so we never apply a country accessor to a non-country root.
#   - no `is_player` anywhere near a non-country root.
#
# After a playtest: grep debug.log for SNW_PROBE to see which on_actions
# fired, which scope names exist on each, and which countries they name.

# ---------------------------------------------------------------------------
# country_swayed: watched vs unwatched  (added 2026-09-15)
# ---------------------------------------------------------------------------
#
# The 1836-1846 census measured 185 player-visible `country_swayed` firings --
# by a wide margin the largest instrumented feed item in ten years. But the
# census records only key and date, so it could not say WHO was swayed, and
# the user's question -- "is this watched countries or noise?" -- was
# unanswerable from it.
#
# The dates already reframed it: all 185 fell in Aug-Oct 1845, i.e. one
# diplomatic play, not chronic drip. That matters, because a burst during a
# war the player is fighting is arguably wanted, while the same volume spread
# over a decade would not be. Before demoting it, we need the split.
#
# Root = Diplomatic Play (vanilla's own comment, 00_code_on_actions.txt at
# on_country_swayed). The play's named scopes are NOT confirmed for this
# specific on_action -- 03_'s scope:actor / scope:target are confirmed on
# OTHER diplo-play on_actions, and this project has a live runtime error on
# record from generalising exactly that way (the `_third_party_` family in
# 06_). So every scope here is reached through the optional operator `?=`,
# which skips silently when unbound and costs nothing when wrong.
#
# Each candidate is logged separately so one run names which scope is actually
# bound, rather than leaving a blank result ambiguous.
on_country_swayed = {
	on_actions = {
		smart_notifications_probe_country_swayed
	}
}

smart_notifications_probe_country_swayed = {
	effect = {
		debug_log = "SNW_SWAY|firing|[TimeKeeper.GetCurrentDate.GetString]"
		scope:actor ?= {
			debug_log = "SNW_SWAY|scope=actor|bound|[THIS.GetCountry.GetNameNoFormatting]"
			if = {
				limit = { smart_notifications_is_watched = yes }
				debug_log = "SNW_SWAY|scope=actor|watched"
			}
			else = {
				debug_log = "SNW_SWAY|scope=actor|unwatched"
			}
		}
		scope:target ?= {
			debug_log = "SNW_SWAY|scope=target|bound|[THIS.GetCountry.GetNameNoFormatting]"
			if = {
				limit = { smart_notifications_is_watched = yes }
				debug_log = "SNW_SWAY|scope=target|watched"
			}
			else = {
				debug_log = "SNW_SWAY|scope=target|unwatched"
			}
		}
		# ADDED 2026-09-15, and it is the one that actually matters. Vanilla's
		# own loc for this notification reads
		#   "[SCOPE.sC('recipient').GetName] swayed"
		# so the country the notification is ABOUT is `recipient` -- not
		# `target`, which is the diplomatic play's target. The first version of
		# this probe measured actor and target and missed the subject entirely,
		# which produced a watched/unwatched split for the wrong country.
		scope:recipient ?= {
			debug_log = "SNW_SWAY|scope=recipient|bound|[THIS.GetCountry.GetNameNoFormatting]"
			if = {
				limit = { smart_notifications_is_watched = yes }
				debug_log = "SNW_SWAY|scope=recipient|watched"
			}
			else = {
				debug_log = "SNW_SWAY|scope=recipient|unwatched"
			}
		}
	}
}
