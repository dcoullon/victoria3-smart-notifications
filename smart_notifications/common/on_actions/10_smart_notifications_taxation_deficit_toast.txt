# Smart Notifications mod: one-time toast per state that enters a
# taxation deficit, independent of the persistent
# smart_notifications_taxation_deficit_alert's dismiss state (see
# common/alert_types/01_smart_notifications_alerts.txt).
#
# ADDED 2026-09-09 per the user, after confirming the exact same
# dismiss-hides-new-instances gap the law commitment toast was built for
# also applies here: the alert is grouped/aggregated
# (alert_group = smart_notifications_taxation_deficit_states) and its
# dismissal is edge-triggered on the alert's OWN valid (any_scope_state
# in deficit) going false->true -- if State A's deficit already showed
# and got dismissed, and a DIFFERENT state B later develops a deficit
# while A's is still ongoing, "some state has a deficit" never flips
# false->true again, so the alert never resurfaces for the new state B.
#
# States are tracked individually via a country-scope VARIABLE LIST
# holding event-target references to the actual state objects
# (add_to_variable_list / is_target_in_variable_list / remove_list_variable
# -- confirmed real via script_docs effects.log/triggers.log), so one
# shared message definition suffices, exactly like vanilla's own
# `type = state` messages (e.g. colony_created) do -- confirmed via
# localization/english/interfaces_l_english.yml's own
# `[SCOPE.GetRootScope.GetState.GetName]` usage in exactly that message's
# loc, the SAME cast chain our own taxation_deficit_alert's loc already
# uses successfully (confirmed working live).
#
# CAP -- REDESIGNED 2026-09-09, FOURTH ITERATION, per the user's own
# precise spec after the third (lifetime-total) version missed the
# actual intent. The real model: each monthly pulse is its own "round".
# Within a round, every state newly crossing into deficit gets marked
# HANDLED immediately, whether or not it actually gets a toast -- only
# up to 3 of that round's newly-crossing states get the actual toast,
# but ALL of them stop being "eligible" the moment the round is
# evaluated, so the other 17 (in a 20-state burst) never trickle out one
# by one over the following months. A state stays marked until it
# RECOVERS (exits deficit) and only becomes eligible again if it
# independently re-enters deficit later -- at which point it's part of
# whatever round that re-entry happens in, with its OWN fresh 3-toast
# budget, separate from any earlier round's.
#
# This needs THREE passes per pulse, not one:
#   A) ordered_scope_state, max = 3: the already-proven select+toast+mark
#      mechanism from the previous iteration, unchanged in shape.
#   B) every_scope_state, uncapped: marks any REMAINING newly-crossing
#      states (beyond A's 3) as handled too, but does NOT post a
#      notification for them. Same exclusion logic as A
#      (NOT already marked), so A's own marks (already committed by the
#      time B runs, since B is a separate, later top-level effect, not
#      nested inside A's own iteration) are correctly seen -- this is
#      NOT the same mid-loop-mutation-invisibility bug found earlier;
#      that was specifically about ONE iterator not seeing ITS OWN
#      in-progress mutations, not about two separate, sequential
#      effects seeing each other's completed ones.
#   C) every_scope_state, uncapped: the recover/re-enter reset, restored
#      from the version before the (wrong) lifetime-cap redesign -- a
#      state that's no longer incorporated-and-in-deficit gets
#      unmarked, so a later independent re-entry is treated as a new
#      round with its own budget.
#
# The membership-check scope bug found live 2026-09-09 (THIS inside an
# X_scope_state iterator's own `limit` is the candidate STATE, not the
# country -- the list lives on the country, so the check must re-scope
# via `THIS.owner` first) applies to A and B identically; both use the
# same corrected pattern.
on_monthly_pulse_country = {
	on_actions = {
		smart_notifications_taxation_deficit_toast_check
	}
}

smart_notifications_taxation_deficit_toast_check = {
	effect = {
		if = {
			limit = { is_player = yes }

			# A) Toast + mark the first 3 newly-crossing states this round.
			ordered_scope_state = {
				limit = {
					is_incorporated = yes
					tax_capacity < tax_capacity_usage
					NOT = {
						THIS.owner = {
							is_target_in_variable_list = {
								name = smart_notifications_notified_deficit_states
								target = prev
							}
						}
					}
				}
				order_by = { value = 1 }
				max = 3
				check_range_bounds = no
				THIS.owner = {
					add_to_variable_list = {
						name = smart_notifications_notified_deficit_states
						target = prev
					}
					prev = {
						post_notification = smart_notifications_taxation_deficit_toast
					}
					debug_log = "SNW_TAX_TOAST|fired|date=[TimeKeeper.GetCurrentDate.GetString]"
				}
			}

			# B) Silently mark any further newly-crossing states this same
			# round (beyond A's 3) as handled -- no toast, but they stop
			# being "eligible" immediately rather than surfacing in a
			# later pulse.
			every_scope_state = {
				limit = {
					is_incorporated = yes
					tax_capacity < tax_capacity_usage
					NOT = {
						THIS.owner = {
							is_target_in_variable_list = {
								name = smart_notifications_notified_deficit_states
								target = prev
							}
						}
					}
				}
				THIS.owner = {
					add_to_variable_list = {
						name = smart_notifications_notified_deficit_states
						target = prev
					}
					debug_log = "SNW_TAX_TOAST|marked_silently|date=[TimeKeeper.GetCurrentDate.GetString]"
				}
			}

			# C) Recover/re-enter reset: a state no longer in deficit is
			# unmarked, so a later independent re-entry starts a fresh
			# round with its own 3-toast budget.
			every_scope_state = {
				limit = {
					NOT = {
						AND = {
							is_incorporated = yes
							tax_capacity < tax_capacity_usage
						}
					}
				}
				THIS.owner = {
					if = {
						limit = {
							is_target_in_variable_list = {
								name = smart_notifications_notified_deficit_states
								target = prev
							}
						}
						remove_list_variable = {
							name = smart_notifications_notified_deficit_states
							target = prev
						}
						debug_log = "SNW_TAX_TOAST|reset|date=[TimeKeeper.GetCurrentDate.GetString]"
					}
				}
			}
		}
	}
}
