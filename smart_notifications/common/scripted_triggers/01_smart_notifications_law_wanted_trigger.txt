# Smart Notifications mod: shared "does this law's next checkpoint have
# Success beating Stall" check, used by the law commitment alert, its
# temporary diagnostic probe, and the new dynamic tooltip list builder --
# see common/alert_types/01_smart_notifications_alerts.txt's own header
# comment for the full reasoning (no single trigger compares two chances
# directly, so this sweeps a 1-point grid, 99 steps, testing whether some
# threshold sits strictly between the two). Factored out 2026-09-08 once a
# third consumer needed the identical logic -- keeping it in one place
# instead of tripling ~400 lines of near-identical script.
#
# Called as `smart_notifications_law_ready_to_enact = yes` from a context
# where THIS is the law being checked (e.g. inside `any_law`/`every_law`).
smart_notifications_law_ready_to_enact = {
	THIS.owner = {
		OR = {
			AND = {
				enactment_chance_for_law = { target = prev.type  value > 0.01 }
				NOT = { stall_chance_for_law = { target = prev.type  value > 0.01 } }
			}
			AND = {
				enactment_chance_for_law = { target = prev.type  value > 0.02 }
				NOT = { stall_chance_for_law = { target = prev.type  value > 0.02 } }
			}
			AND = {
				enactment_chance_for_law = { target = prev.type  value > 0.03 }
				NOT = { stall_chance_for_law = { target = prev.type  value > 0.03 } }
			}
			AND = {
				enactment_chance_for_law = { target = prev.type  value > 0.04 }
				NOT = { stall_chance_for_law = { target = prev.type  value > 0.04 } }
			}
			AND = {
				enactment_chance_for_law = { target = prev.type  value > 0.05 }
				NOT = { stall_chance_for_law = { target = prev.type  value > 0.05 } }
			}
			AND = {
				enactment_chance_for_law = { target = prev.type  value > 0.06 }
				NOT = { stall_chance_for_law = { target = prev.type  value > 0.06 } }
			}
			AND = {
				enactment_chance_for_law = { target = prev.type  value > 0.07 }
				NOT = { stall_chance_for_law = { target = prev.type  value > 0.07 } }
			}
			AND = {
				enactment_chance_for_law = { target = prev.type  value > 0.08 }
