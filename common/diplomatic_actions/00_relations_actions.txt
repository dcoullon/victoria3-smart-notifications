# MOD OVERRIDE of vanilla common/diplomatic_actions/00_relations_actions.txt
# (Victoria 3 1.13.11). Byte-for-byte vanilla EXCEPT for the two `effect`
# blocks marked "SMART NOTIFICATIONS" below. Vanilla defines exactly two
# actions in this file -- increase_relations and damage_relations -- which
# is precisely and only the pair this mod needs to tag, so overriding it
# costs nothing else.
#
# WHY THIS FILE EXISTS AT ALL. This mod wants to demote a routine
# relations change to the notification feed while keeping every other
# diplomatic action a toast. That decision has to be made in
# `on_diplomatic_action` (see 06_smart_notifications_diplomatic_action_filtering.txt),
# and at that instant the engine gives no way whatsoever to ask what kind
# of action just fired -- confirmed at length, see docs/engine-notes.md
# § The pact for a diplomatic action does not exist yet. So instead of
# asking "what type was this action?", we attach script to the two action
# types themselves: the action's own definition obviously knows what it
# is. The effect flags the target country, and the notification handler
# reads the flag.
#
# UNPROVEN MECHANISM, DELIBERATELY INSTRUMENTED. `effect = {}` ("Effect of
# action on execution", root = action initiator) is documented in the
# game's own common/diplomatic_actions/diplomatic_action.md schema, but
# **zero** vanilla diplomatic actions use one -- there is no in-game
# precedent to copy, which is exactly the situation this project's rules
# say to test rather than trust (CLAUDE.md § Engine & Syntax Rules). Two
# things are unknown until a live run: whether the block fires at all,
# and whether it fires BEFORE `on_diplomatic_action` (needed -- the flag
# has to exist by the time the notification is posted) or after (useless
# as written). The SNW_ROUTINE_FLAG debug lines below and the matching
# SNW_ROUTINE_READ lines in the on_action exist to answer both from a
# single session, by their order in debug.log. Delete the logging once
# answered.
#
# MAINTENANCE COST, accepted knowingly: this file will conflict with any
# other mod that overrides the same file, and it can silently drift when
# Paradox patches vanilla's version. Re-diff it against vanilla on each
# game update -- reference/vanilla holds the pristine copy for exactly
# this. Tracked in TODO.md.

increase_relations = {
	groups= {
		general
	}

	requires_approval = no
	show_confirmation_box = no

	selectable = {
		NOT = { is_country_type = decentralized }
	}

	potential = {
		scope:target_country = {
			NOT = { is_country_type = decentralized }
		}
	}

	possible = {
		NOT = { has_diplomatic_pact = { who = scope:target_country type = damage_relations } }
	}


	# SMART NOTIFICATIONS -- the only modification to this file. Flags the
	# target so on_diplomatic_action can recognise this as a routine
	# relations change. `days = 2` is a safety net, not the mechanism: the
	# on_action clears the flag as soon as it reads it, and the expiry only
	# matters if the notification never arrives to clear it (in which case
	# a stale flag must not linger and mis-demote a later, important
	# action against the same country).
	effect = {
		# Static line FIRST, dynamic line second, on purpose: if the block
		# fires but the bracket text is wrong, the dynamic line vanishes
		# silently ("Data error in loc string") and would look identical to
		# the block never firing at all. The static line can't fail, so the
		# two failure modes stay distinguishable. Same lesson as
		# docs/engine-notes.md § THIS needs a cast.
		debug_log = "SNW_ROUTINE_FLAG|fired|type=increase_relations"
		scope:target_country = {
			set_variable = { name = snw_routine_relations_action days = 2 }
		}
		debug_log = "SNW_ROUTINE_FLAG|set|type=increase_relations|initiator=[THIS.GetCountry.GetNameNoFormatting]|target=[SCOPE.sC('target_country').GetNameNoFormatting]"
	}

	pact = {
		cost = 150
		is_two_sided_pact = no
		show_in_outliner = yes

		max_target_involvement = 1000
		target_involvement_applies_to = source_country

		relations_progress_per_day = 3 # At 100, +1 relations
		relations_improvement_max = 50

			requirement_to_maintain = {
			trigger = {
				has_diplomatic_relevance = scope:target_country
			}
		}

		requirement_to_maintain = {
			trigger = {
				custom_tooltip = {
					text = "DIPLOMATIC_PACT_REQUIREMENT_AT_PEACE"
					NOT = { has_war_with = scope:target_country }
				}
			}
		}

		requirement_to_maintain = {
			trigger = {
				custom_tooltip = {
					text = "DIPLOMATIC_PACT_REQUIREMENT_NOT_DECENTRALIZED"
					NOT = { is_country_type = decentralized }
					scope:target_country = {
						NOT = { is_country_type = decentralized }
					}
				}
			}
		}

		requirement_to_maintain = {
			trigger = {
				custom_tooltip = {
					text = DIPLO_ACTION_RELATIONS_AMICABLE_OR_GREATER_NOT
					NOT = {
						scope:target_country.relations:root >= relations_threshold:amicable
					}
				}
			}
		}

		requirement_to_maintain = {
			trigger = {
				NOT = { has_diplomats_expelled = scope:target_country }
			}
		}

		requirement_to_maintain = {
			trigger = {
				NOT = { scope:target_country = { has_diplomats_expelled = root } }
			}
		}
	}

	ai = {
		max_influence_spending_fraction = 0.25

		evaluation_chance = {
			value = 0.5
		}

		propose_score = {
			value = 5

			if = {
				limit = {
					has_attitude = {
						who = scope:target_country
						attitude = protective
					}
				}

				add = 5
			}

			if = {
				limit = {
					has_attitude = {
						who = scope:target_country
						attitude = conciliatory
					}
				}

				add = 20
			}

			if = {
				limit = {
					has_journal_entry = je_hawaiian_interest
					c:HAW ?= {
						this = scope:target_country
					}
				}
				add = 20
			}

			add = {
				value = scope:target_country.country_rank
				multiply = 3
			}

			if = {
				limit = {
					OR = {
						is_subject_of = scope:target_country
						scope:target_country = { is_subject_of = root }
					}
				}

				multiply = 2
			}
		}

		will_propose = {
			OR = {
				AND = {
					has_attitude = {
						who = scope:target_country
						attitude = cooperative
					}
					country_rank <= scope:target_country.country_rank
				}
				has_attitude = {
					who = scope:target_country
					attitude = conciliatory
				}
				has_attitude = {
					who = scope:target_country
					attitude = genial
				}
				has_attitude = {
					who = scope:target_country
					attitude = protective
				}
			}
		}

		will_break = {
			OR = {
				NOR = {
					has_attitude = {
						who = scope:target_country
						attitude = genial
					}
					has_attitude = {
						who = scope:target_country
						attitude = conciliatory
					}
					has_attitude = {
						who = scope:target_country
						attitude = cooperative
					}
					has_attitude = {
						who = scope:target_country
						attitude = protective
					}
				}
				AND = {
					scope:target_country.relations:root >= relations_threshold:amicable
					OR = {
						country_rank >= rank_value:minor_power
						NOT = {
							has_attitude = {
								who = scope:target_country
								attitude = genial
							}
						}
					}
				}
			}

		}
	}
}

damage_relations = {
	groups= {
		general
	}

	requires_approval = no
	show_confirmation_box = no

	selectable = {
		NOT = { is_country_type = decentralized }
	}

	potential = {
		scope:target_country = {
			NOT = { is_country_type = decentralized }
		}
	}

	possible = {
		NOT = { has_diplomatic_pact = { who = scope:target_country type = increase_relations } }
	}


	# SMART NOTIFICATIONS -- the only modification to this file. Flags the
	# target so on_diplomatic_action can recognise this as a routine
	# relations change. `days = 2` is a safety net, not the mechanism: the
	# on_action clears the flag as soon as it reads it, and the expiry only
	# matters if the notification never arrives to clear it (in which case
	# a stale flag must not linger and mis-demote a later, important
	# action against the same country).
	effect = {
		# Static line FIRST, dynamic line second, on purpose: if the block
		# fires but the bracket text is wrong, the dynamic line vanishes
		# silently ("Data error in loc string") and would look identical to
		# the block never firing at all. The static line can't fail, so the
		# two failure modes stay distinguishable. Same lesson as
		# docs/engine-notes.md § THIS needs a cast.
		debug_log = "SNW_ROUTINE_FLAG|fired|type=damage_relations"
		scope:target_country = {
			set_variable = { name = snw_routine_relations_action days = 2 }
		}
		debug_log = "SNW_ROUTINE_FLAG|set|type=damage_relations|initiator=[THIS.GetCountry.GetNameNoFormatting]|target=[SCOPE.sC('target_country').GetNameNoFormatting]"
	}

	pact = {
		cost = 100
		is_hostile = yes
		is_two_sided_pact = no
		infamy_affects_maintenance = no
		show_in_outliner = yes

		relations_progress_per_day = -3 # At -100, -1 relations
		relations_improvement_min = -50

		requirement_to_maintain = {
			trigger = {
				has_diplomatic_relevance = scope:target_country
			}
		}

		requirement_to_maintain = {
			trigger = {
				custom_tooltip = {
					text = DIPLO_ACTION_RELATIONS_COLD_OR_LESS_NOT
					scope:target_country.relations:root > relations_threshold:cold
				}
			}
		}

		requirement_to_maintain = {
			trigger = {
				custom_tooltip = {
					text = "DIPLOMATIC_PACT_REQUIREMENT_NOT_DECENTRALIZED"
					NOT = { is_country_type = decentralized }
					scope:target_country = {
						NOT = { is_country_type = decentralized }
					}
				}
			}
		}
	}

	ai = {
		max_influence_spending_fraction = 0.15

		evaluation_chance = {
			value = 0.5
		}

		propose_score = {
			value = 10

			if = {
				limit = {
					OR = {
						has_attitude = {
							who = scope:target_country
							attitude = belligerent
						}
						has_attitude = {
							who = scope:target_country
							attitude = domineering
						}
						has_attitude = {
							who = scope:target_country
							attitude = rebellious
						}
					}
				}

				add = 10
			}

			if = {
				limit = {
					OR = {
						is_subject_of = scope:target_country
						scope:target_country = { is_subject_of = root }
					}
				}

				multiply = 5
			}

			if = {
				limit = {
					AND = {
						has_radical_leftist_government = no
						scope:target_country = {
							has_radical_leftist_government = yes
						}
					}

				}
				multiply = 2
			}

			if = {
				limit = {
					AND = {
						has_radical_leftist_economy = no
						scope:target_country = {
							has_radical_leftist_economy = yes
						}
					}

				}
				multiply = 2
			}
		}

		will_propose = {
			OR = {
				AND = {
					has_attitude = {
						who = scope:target_country
						attitude = antagonistic
					}
					scope:target_country.relations:root > 0
				}
				has_attitude = {
					who = scope:target_country
					attitude = belligerent
				}
				has_attitude = {
					who = scope:target_country
					attitude = domineering
				}
				has_attitude = {
					who = scope:target_country
					attitude = rebellious
				}
			}
			scope:target_country.relations:root > relations_threshold:poor
		}

		will_break = {
			OR = {
				NOR = {
					has_attitude = {
						who = scope:target_country
						attitude = belligerent
					}
					has_attitude = {
						who = scope:target_country
						attitude = domineering
					}
					has_attitude = {
						who = scope:target_country
						attitude = antagonistic
					}
					has_attitude = {
						who = scope:target_country
						attitude = rebellious
					}
				}
				scope:target_country.relations:root <= relations_threshold:poor
			}
		}
	}
}
