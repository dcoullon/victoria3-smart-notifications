# Smart Notifications mod: custom alert-type entries.
#
# See docs/engine-notes.md "Two entirely separate notification systems" --
# alert_types (unlike common/messages/, which is a full-file-path override)
# are additive across files in this folder, so this file only needs OUR new
# entries, not a copy of vanilla's 00_alert_types.txt.

smart_notifications_amendment_repealable_alert = {
	script_context = player_country
	texture = "gfx/interface/icons/notification_icons/laws.dds"

	# amendment_can_be_repealed is scoped to `amendment`, and
	# any_scope_amendment is scoped to `law` (not `country`) -- both
	# confirmed via the real script_docs dump (docs/engine-notes.md), so
	# any_active_law has to be the outer iterator to reach it from country
	# scope.
	valid = {
		any_active_law = {
			any_scope_amendment = {
				amendment_can_be_repealed = yes
			}
		}
	}

	# "politics|default" is a confirmed vanilla open_panel value (used by
	# political_lobby-adjacent alerts); a law/amendment-specific tab name
	# wasn't found in the dump, so using the panel itself rather than
	# guessing a tab identifier that might not exist.
	open_panel = politics|default
	type = important_action
	color = green
}

smart_notifications_agitator_invite_available_alert = {
	script_context = player_country
	texture = "gfx/interface/icons/character_role_icons/agitator.dds"

	# All three triggers below confirmed real via the game's own
	# script_docs triggers.log and cross-referenced against vanilla's own
	# `invite_exile` character interaction
	# (common/character_interactions/00_character_interactions.txt), which
	# this alert is meant to mirror the eligibility of exactly:
	# - has_dlc_feature = agitators -- confirmed real (used by
	#   grant_leadership_to_agitator/grant_command_to_agitator in the same
	#   file); agitators are gated behind the Voice of the People
	#   Immersion Pack, not base game content.
	# - empty_agitator_slots -- country-scope trigger, "checks number of
	#   empty agitator slots in a country".
	# - any_character_in_exile_pool / can_agitate -- any_character_in_exile_pool
	#   is a global iterator (no scope needed) over the exile pool;
	#   can_agitate = <country> checks whether the scoped character is
	#   eligible to agitate that country. This is the exact same pairing
	#   `invite_exile`'s own `possible` block uses
	#   (`can_agitate = scope:actor`), so this reuses vanilla's real
	#   eligibility gate instead of guessing a new one.
	valid = {
		has_dlc_feature = agitators
		empty_agitator_slots >= 1
		any_character_in_exile_pool = {
			can_agitate = ROOT
		}
	}

	# UNCONFIRMED, needs in-game verification: vanilla's `invite_exile`
	# interaction also carries its own cooldown (cooldown = { days =
	# normal_modifier_time }, 1825 days = 5 years -- exactly the ~5-year
	# figure the user described) after a successful invite. That's a
	# native per-interaction cooldown with no documented trigger to query
	# it directly from script (see docs/engine-notes.md). The working
	# assumption is that this doesn't matter in practice: the slot the
	# invited agitator now occupies stays filled for ~5-7.5 years
	# (add_career_length months = 60-90 on invite), so empty_agitator_slots
	# should already read 0 for the whole cooldown window on its own,
	# naturally keeping this alert quiet without needing to check the
	# cooldown separately. If a session ever shows this alert firing while
	# the Invite Exile interaction is still greyed out on cooldown, that
	# assumption was wrong and needs revisiting.
	open_panel = politics|default
	type = important_action
	color = green
}

smart_notifications_taxation_deficit_alert = {
	texture = "gfx/interface/icons/state_status_icons/state_tax_capacity.dds"
	script_context = player_state

	# Base version only, per TODO.md -- the "only alert if fixing it pays
	# for itself" refinement is scoped separately, not attempted here.
	#
	# Trigger reused verbatim from vanilla's OWN commented-out, never-shipped
	# `insufficient_tax_capacity_alert`
	# (common/alert_types/00_alert_types.txt, filed under "Hidden by
	# default") rather than guessing a condition ourselves -- Paradox
	# already worked out the right trigger for exactly this alert and
	# simply never turned it on. `tax_capacity`/`tax_capacity_usage` are
	# both confirmed real state-scope triggers (script_docs
	# triggers.log); a state whose usage exceeds its capacity has
	# "Insufficient Taxation Capacity" per the game's own tooltip wording
	# (STATE_TAX_CAPACITY_INSUFFICIENT_LONG,
	# localization/english/interfaces_l_english.yml) -- this is the
	# "taxation deficit" the user described. `is_incorporated = yes`
	# (also confirmed real, state scope) excludes unincorporated states,
	# matching vanilla's own version.
	valid = {
		tax_capacity < tax_capacity_usage
		is_incorporated = yes
	}

	# alert_group, not just type = important_action alone -- per the
	# alert_types header comment, an important_action with no alert_group
	# never groups with others of the same type, and this alert can fire
	# once per state, so multiple simultaneous deficits would otherwise
	# render as separate important-action entries instead of one stacked
	# group (matching vanilla's own choice to group this alert by state).
	open_panel = states_panel
	type = important_action
	color = green
	alert_group = smart_notifications_taxation_deficit_states
}

smart_notifications_law_commitment_alert = {
	script_context = player_country
	texture = "gfx/interface/icons/notification_icons/laws.dds"

	# The biggest/highest-risk item in TODO.md's sized backlog -- see
	# common/scripted_guis/smart_notifications_law_notify_sgui.txt (the
	# opt-in checkbox toggle) and
	# gui/politics_panel_change_law.gui (where the checkbox actually
	# lives, on the law detail panel) for the two other pieces this
	# depends on.
	#
	# REVISED 2026-09-08 per the user: this must flag a SPECIFIC law the
	# player wants and tell them when IT now has more support than
	# pushback -- not "notify me about whatever I happen to be enacting
	# right now" (the first version built, which the user correctly
	# rejected as useless for deciding whether to START enacting
	# something).
	#
	# REVISED 2026-09-08: the "wanted" flag couldn't live on the `law`
	# scope itself -- error.log confirmed, for every law type tested,
	# "has_variable trigger [ This scope doesn't support variables.
	# Scope: Law ... ]" -- `law` scope categorically cannot hold
	# variables in this engine. Moved to a country-scope, per-law-type
	# static variable instead; see
	# common/scripted_triggers/01_smart_notifications_law_wanted_trigger.txt
	# (the generated law_type <-> variable_name mapping, shared with the
	# check_sgui's own `is_valid`).
	#
	# REVISED AGAIN 2026-09-08, per the user: "ready to activate" means
	# the next checkpoint's Success chance beating its Stall chance (the
	# same comparison the law list itself displays), not an absolute
	# ">50% overall" threshold -- a law showing Success 43% / Stall 30%
	# is "success > stall" but was reading false against a flat 0.5.
	# `smart_notifications_law_ready_to_enact` (same trigger file as
	# above) is the shared "does this law's next checkpoint favor
	# success" check -- see its own header comment for why it needs a
	# 99-point threshold sweep instead of a single comparison (no trigger
	# compares two chances directly, and neither can be pulled out as a
	# raw number for script math). Factored out of this file 2026-09-08
	# once the new tooltip list builder below needed the identical logic
	# a third time.
	#
	# All confirmed real via script_docs triggers.log/event_targets.log:
	# - any_law -- "iterate through all laws in a country" (country scope,
	#   target = law), unlike any_active_law which only sees the one
	#   active law per group -- this needs to see every AVAILABLE law
	#   option, flagged or not, active or not.
	# - smart_notifications_law_matches_wanted_flag -- shared scripted
	#   trigger, true if THIS (the iterated law) matches one of the
	#   country's flagged law types.
	# - can_be_enacted = yes (law scope) -- "checks if a law could be
	#   enacted by its country, considering its current situation";
	#   excludes the already-active law in its group and anything
	#   currently blocked, so this alert self-cleans once a flagged law
	#   passes or becomes unavailable, with no separate reset needed.
	valid = {
		any_law = {
			smart_notifications_law_matches_wanted_flag = yes
			can_be_enacted = yes
			smart_notifications_law_ready_to_enact = yes
		}
	}

	# "politics|laws" is the confirmed real tab identifier for the Laws
	# tab (gui/politics_panel.gui: `InformationPanel.SelectTab('laws')`).
	# CHECKED 2026-09-08, per the user's ask for a direct deep link to the
	# specific law/group: the real native function for this,
	# `InformationPanelBar.OpenChangeLaw(Law.GetGroup)`
	# (gui/politics_panel_types.gui, the law list's own row onclick), is a
	# GUI-only action taking a live object argument -- alert_types'
	# `open_panel`/`open_popup` only accept a static "panel[|tab]" string
	# (confirmed via the alert_types header comment and every vanilla
	# example), with no way to parameterize it with a specific group at
	# script time. No deeper link is possible here; this opens the Laws
	# overview (where every group is at least visible to click into), and
	# the alert's own text below is how the player learns WHICH group.
	open_panel = politics|laws
	type = important_action
	color = green
}

# Smart Notifications mod: dynamic tooltip list builder for the alert
# above, ADDED 2026-09-08 per the user (originally shipped with static
# text; the user asked for the specific law name and group, since
# clicking the alert alone doesn't highlight anything -- see the
# open_panel comment above for why a direct link isn't possible).
#
# Real, documented vanilla technique (common/scripted_guis/scripted_guis.md
# "Using SGUIs to build lists in loc"): a scripted GUI's `effect` can call
# `custom_tooltip = "LOC_KEY"` once per matching item, and a loc string
# elsewhere can render the whole built list live via
# `[GetScriptedGui('key').ExecuteTooltip(GuiScope.SetRoot(...).End)]` --
# evaluated fresh every time the tooltip renders, not baked into a slow-
# updating variable, so there's no staleness/lag concern the way a
# monthly-pulse-computed value would have.
smart_notifications_law_commitment_list_sgui = {
	scope = country

	effect = {
		if = {
			limit = {
				any_law = {
					smart_notifications_law_matches_wanted_flag = yes
					can_be_enacted = yes
					smart_notifications_law_ready_to_enact = yes
				}
			}
			every_law = {
				limit = {
					smart_notifications_law_matches_wanted_flag = yes
					can_be_enacted = yes
					smart_notifications_law_ready_to_enact = yes
				}
				custom_tooltip = "smart_notifications_law_commitment_list_entry"
			}
		}
		else = {
			custom_tooltip = "smart_notifications_law_commitment_list_empty"
		}
	}

	# Per CLAUDE.md: scripted GUIs must never be available to the AI.
	ai_is_valid = { always = no }
	ai_chance = { base = 0 }
}
