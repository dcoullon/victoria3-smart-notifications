# Smart Notifications mod: custom alert-type entries.
#
# See docs/engine-notes.md "Two entirely separate notification systems" --
# alert_types (unlike common/messages/, which is a full-file-path override)
# are additive across files in this folder, so this file only needs OUR new
# entries, not a copy of vanilla's 00_alert_types.txt.

smart_notifications_amendment_repealable_alert = {
	script_context = player_country
	texture = "gfx/interface/icons/notification_icons/laws.dds"

	# amendment_can_be_repealed is scoped to `amendment`, and
	# any_scope_amendment is scoped to `law` (not `country`) -- both
	# confirmed via the real script_docs dump (docs/engine-notes.md), so
	# any_active_law has to be the outer iterator to reach it from country
	# scope.
	valid = {
		any_active_law = {
			any_scope_amendment = {
				amendment_can_be_repealed = yes
			}
		}
	}

	# "politics|default" is a confirmed vanilla open_panel value (used by
	# political_lobby-adjacent alerts); a law/amendment-specific tab name
	# wasn't found in the dump, so using the panel itself rather than
	# guessing a tab identifier that might not exist.
	open_panel = politics|default
	type = important_action
	color = green
}

smart_notifications_agitator_invite_available_alert = {
	script_context = player_country
	texture = "gfx/interface/icons/character_role_icons/agitator.dds"

	# All three triggers below confirmed real via the game's own
	# script_docs triggers.log and cross-referenced against vanilla's own
	# `invite_exile` character interaction
	# (common/character_interactions/00_character_interactions.txt), which
	# this alert is meant to mirror the eligibility of exactly:
	# - has_dlc_feature = agitators -- confirmed real (used by
	#   grant_leadership_to_agitator/grant_command_to_agitator in the same
	#   file); agitators are gated behind the Voice of the People
	#   Immersion Pack, not base game content.
	# - empty_agitator_slots -- country-scope trigger, "checks number of
	#   empty agitator slots in a country".
	# - any_character_in_exile_pool / can_agitate -- any_character_in_exile_pool
	#   is a global iterator (no scope needed) over the exile pool;
	#   can_agitate = <country> checks whether the scoped character is
	#   eligible to agitate that country. This is the exact same pairing
	#   `invite_exile`'s own `possible` block uses
	#   (`can_agitate = scope:actor`), so this reuses vanilla's real
	#   eligibility gate instead of guessing a new one.
	valid = {
		has_dlc_feature = agitators
		empty_agitator_slots >= 1
		any_character_in_exile_pool = {
			can_agitate = ROOT
		}
	}

	# UNCONFIRMED, needs in-game verification: vanilla's `invite_exile`
	# interaction also carries its own cooldown (cooldown = { days =
	# normal_modifier_time }, 1825 days = 5 years -- exactly the ~5-year
	# figure the user described) after a successful invite. That's a
	# native per-interaction cooldown with no documented trigger to query
	# it directly from script (see docs/engine-notes.md). The working
	# assumption is that this doesn't matter in practice: the slot the
	# invited agitator now occupies stays filled for ~5-7.5 years
	# (add_career_length months = 60-90 on invite), so empty_agitator_slots
	# should already read 0 for the whole cooldown window on its own,
	# naturally keeping this alert quiet without needing to check the
	# cooldown separately. If a session ever shows this alert firing while
	# the Invite Exile interaction is still greyed out on cooldown, that
	# assumption was wrong and needs revisiting.
	open_panel = politics|default
	type = important_action
	color = green
}

smart_notifications_taxation_deficit_alert = {
	texture = "gfx/interface/icons/state_status_icons/state_tax_capacity.dds"
	script_context = player_state

	# Base version only, per TODO.md -- the "only alert if fixing it pays
	# for itself" refinement is scoped separately, not attempted here.
	#
	# Trigger reused verbatim from vanilla's OWN commented-out, never-shipped
	# `insufficient_tax_capacity_alert`
	# (common/alert_types/00_alert_types.txt, filed under "Hidden by
	# default") rather than guessing a condition ourselves -- Paradox
	# already worked out the right trigger for exactly this alert and
	# simply never turned it on. `tax_capacity`/`tax_capacity_usage` are
	# both confirmed real state-scope triggers (script_docs
	# triggers.log); a state whose usage exceeds its capacity has
	# "Insufficient Taxation Capacity" per the game's own tooltip wording
	# (STATE_TAX_CAPACITY_INSUFFICIENT_LONG,
	# localization/english/interfaces_l_english.yml) -- this is the
	# "taxation deficit" the user described. `is_incorporated = yes`
	# (also confirmed real, state scope) excludes unincorporated states,
	# matching vanilla's own version.
	valid = {
		tax_capacity < tax_capacity_usage
		is_incorporated = yes
	}

	# alert_group, not just type = important_action alone -- per the
	# alert_types header comment, an important_action with no alert_group
	# never groups with others of the same type, and this alert can fire
	# once per state, so multiple simultaneous deficits would otherwise
	# render as separate important-action entries instead of one stacked
	# group (matching vanilla's own choice to group this alert by state).
	open_panel = states_panel
	type = important_action
	color = green
	alert_group = smart_notifications_taxation_deficit_states
}
