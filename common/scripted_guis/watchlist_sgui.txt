# Smart Notifications mod: Country Watchlist (Phase 3, see TODO.md for the
# full spec). Toggles whether a country is on the player's watchlist.
#
# Data model (TODO.md "Data model — REVISED 2026-09-06"): 4 separate
# provenance flags (great_power/neighbor/rival/manually), not 1, so a bulk
# category action can't silently un-watch a country kept for a different
# reason. This SGUI is the INDIVIDUAL toggle (the country-panel star and,
# later, a row checkbox on the management screen) -- per the spec, that
# means "watched or not, period": clears all four flags to remove, or sets
# just `watched_manually` to add (a star click doesn't know about
# category membership, so a fresh watch via this button is always
# recorded as a manual one).
#
# Structure confirmed against `common/scripted_guis/scripted_guis.md` and
# real working examples (`debug_sguis.txt`'s single-scope toggles).
# CORRECTED 2026-09-07, real bug found live (confirmed by the user: the
# checkbox never toggled, even after a full window close/reopen -- so it
# wasn't a redraw/caching issue, the click was being refused outright).
# Root cause: `is_valid` is not just a display query, it's the actual gate
# that determines whether `.Execute()` is allowed to run at all ("Determines
# whether the SGUI can be used by a player" per scripted_guis.md). Setting
# it to "is this country already watched" meant clicking to WATCH an
# unwatched country was always refused, since is_valid was false for
# exactly the countries you'd want to click.
#
# Fixed by splitting into two SGUIs, the same way vanilla does when a check
# and an action need different validity rules -- confirmed via a real
# working example, `je_meiji_restoration_japanese_emperor_check_sgui`
# (journal_entry_sguis.txt), which is `is_valid`-only with NO effect at
# all, used purely for a `.IsValid()` display check elsewhere.
watchlist_toggle_sgui = {
	scope = country

	# No is_valid here on purpose -- toggling is always a valid action
	# regardless of current watched state, on or off.
	#
	# Used by the Watched and Add a Country rows only (see the three
	# section-specific variants below for Great Powers/Neighbors/Rivals).
	# `watched_manually` is the correct else-branch flag for both: Watched
	# never actually reaches it (every row shown there is already watched,
	# so its checkbox is only ever unchecked, never checked-from-blank),
	# and Add a Country's whole point is manually adding a new one.

	effect = {
		if = {
			limit = {
				OR = {
					has_variable = watched_via_great_power
					has_variable = watched_via_neighbor
					has_variable = watched_via_rival
					has_variable = watched_manually
				}
			}
			remove_variable = watched_via_great_power
			remove_variable = watched_via_neighbor
			remove_variable = watched_via_rival
			remove_variable = watched_manually
		}
		else = {
			set_variable = watched_manually
		}
	}

	# Per CLAUDE.md: scripted GUIs must never be available to the AI.
	ai_is_valid = { always = no }
	ai_chance = { base = 0 }
}

# --- Category-tab row filters (Phase 3 categories, added 2026-09-07) ---
# Each is read-only (is_valid-only, no effect), same shape as
# watchlist_is_watched_check_sgui above, used purely to decide whether a
# given row is visible on a given category sub-tab in
# gui/message_settings.gui.
#
# Great Powers is a TRUE live check (country_rank is absolute, no
# comparison to the player needed) -- browsing shows anyone CURRENTLY a
# Great Power, watched or not, exactly per the original spec.
#
# Neighbors and Rivals -- REWRITTEN 2026-09-07 (second attempt), see the
# bug writeup below.
#
# Attempt 1 (v0.23) used `any_country = { is_player = yes <trigger using
# root> }` to locate the player from inside the trigger. That idiom IS
# real and works in normal effect/on_action script (vanilla uses it in
# common/scripted_buttons/00_balkan_wars_buttons.txt, events/ethiopia.txt,
# events/krakow_events.txt, and this mod's own game-start hook relies on
# it) -- but inside a scripted_gui's `is_valid` it did NOT behave the same
# way, confirmed live: the Neighbors tab listed countries that obviously
# aren't adjacent to the player (Ajuran, Aulihan and other Horn-of-Africa
# minors, in a Portugal game), while the bulk Select All -- which uses the
# player as root directly and never calls `is_player` -- flagged exactly
# the plausible ones. So the displayed list was a strict superset of the
# real neighbours, which is ALSO why "Select All doesn't select all" and
# "Deselect All keeps some rows": both bulk actions were correct, the list
# they were being judged against was not.
#
# Root cause not conclusively isolated (either `is_player` doesn't filter
# or `root` doesn't survive the nested iterator in this context -- the
# evidence fits the former better, since the Rivals tab stayed populated
# rather than going empty). Rather than guess a third time, this rewrite
# ROUTES AROUND the whole uncertain area: a global variable holds the
# player's country, so each row check is one flat trigger evaluated on the
# row's own country -- no `any_country` iterator, no `is_player`, no
# `root` keyword, nothing that depends on scope-nesting semantics inside a
# scripted_gui. Both halves are confirmed vanilla patterns:
# `set_global_variable = { name = x value = ROOT }` storing a country
# scope (common/decisions/01_russia_decisions.txt) and
# `global_var:x ?= THIS` comparing one in a trigger
# (common/diplomatic_plays/00_diplomatic_plays.txt).
#
# The variable is set at game start AND refreshed every month
# (00_smart_notifications_on_actions.txt /
# 02_smart_notifications_truce_tracker.txt), so it self-heals on a save
# that predates it -- but on such a save both lists stay empty until the
# first monthly tick after loading. Every bulk button also refreshes it as
# a side effect, so clicking one fixes the display immediately.
watchlist_is_great_power_check_sgui = {
	scope = country

	is_valid = {
		# Decentralized countries are excluded from the whole Watchlist
		# (user, 2026-09-07): you don't meaningfully interact with them, so
		# watching one is meaningless -- and they were also the countries
		# Select All appeared to "miss", because they are adjacent for
		# display purposes but not picked up the same way by the bulk scan.
		# Excluding them makes the list and the bulk actions agree.
		NOT = { is_country_type = decentralized }
		country_rank >= rank_value:great_power
	}

	ai_is_valid = { always = no }
	ai_chance = { base = 0 }
}

watchlist_is_neighbor_check_sgui = {
	scope = country

	is_valid = {
		# ############################################################
		# KNOWN-GOOD. Confirmed correct in-game by the user 2026-09-07:
		# the resulting list is right for their campaign (a heavily
		# expanded Portugal really is adjacent to Horn-of-Africa minors,
		# which was twice misread as the list being over-inclusive).
		#
		# `root` here is the ROW's country (set by the GUI via
		# InterestingCountryItem.GetCountry.MakeScope) and `this` inside
		# any_country is the player, so this reads "is the player adjacent
		# to this row's country". It must NOT be rewritten to
		# global_var:smart_notifications_player_country -- that makes it
		# "is the player adjacent to the player", which is always false and
		# empties the whole tab. That exact regression shipped twice
		# (v0.28 deliberately, v0.30 by a careless blanket string-replace
		# that clobbered the line it had just restored). Change nothing
		# here without an in-game retest.
		# ############################################################
		# Decentralized countries are excluded from the whole Watchlist
		# (user, 2026-09-07): you don't meaningfully interact with them, so
		# watching one is meaningless -- and they were also the countries
		# Select All appeared to "miss", because they are adjacent for
		# display purposes but not picked up the same way by the bulk scan.
		# Excluding them makes the list and the bulk actions agree.
		NOT = { is_country_type = decentralized }
		any_country = {
			is_player = yes
			is_adjacent_to_country = root
		}
	}

	ai_is_valid = { always = no }
	ai_chance = { base = 0 }
}

watchlist_is_rival_check_sgui = {
	scope = country

	is_valid = {
		# ############################################################
		# KNOWN-GOOD -- same rules as the neighbour check above: `root` is
		# the ROW's country, `this` inside any_country is the player. Do
		# not rewrite `root` here either. Confirmed correct in-game
		# 2026-09-07 (survived the v0.30 regression only because its
		# phrasing happened not to match the blanket replace that broke
		# the neighbour check).
		# ############################################################
		# Decentralized countries are excluded from the whole Watchlist
		# (user, 2026-09-07): you don't meaningfully interact with them, so
		# watching one is meaningless -- and they were also the countries
		# Select All appeared to "miss", because they are adjacent for
		# display purposes but not picked up the same way by the bulk scan.
		# Excluding them makes the list and the bulk actions agree.
		NOT = { is_country_type = decentralized }
		any_country = {
			is_player = yes
			any_rival_country = {
				this ?= root
			}
		}
	}

	ai_is_valid = { always = no }
	ai_chance = { base = 0 }
}

# --- Continent grouping for the Neighbors list (added 2026-09-07) ---
# A colonial-holding player (e.g. Portugal with scattered overseas
# territory) can be genuinely adjacent to dozens of countries across
# totally different regions -- confirmed real, not a bug (see TODO.md).
# Grouped the Neighbors list by continent instead of adding another
# clickable sub-nav layer, using the exact continent split vanilla
# itself uses for `is_country_on_same_continent`
# (common/scripted_triggers/00_geography_triggers.txt): a country's
# capital's state region's `is_in_geographic_region` membership in one
# of the 4 macro-regions vanilla defines (Europe/Americas/Africa/Asia --
# there is no separate Oceania bucket in vanilla's own version of this
# check, so Oceania countries fall under Asia here too, same as vanilla
# does). No player reference needed at all for these -- purely a
# property of the row's own country.
watchlist_is_in_europe_sgui = {
	scope = country

	is_valid = {
		capital = {
			state_region = {
				is_in_geographic_region = geographic_region_europe
			}
		}
	}

	ai_is_valid = { always = no }
	ai_chance = { base = 0 }
}

# --- Category bulk select/deselect -- REWRITTEN 2026-09-07 (third and
# final approach, after the root probes came back) ---
#
# History, so this isn't "fixed" back into a broken form again: the bulk
# buttons never worked, across three rewrites of their effect BODY
# (root -> global variable -> back). The probes settled why: a
# button-triggered scripted GUI DOES run its effect, but the root it is
# handed is not a usable country, so every `root`-dependent limit matched
# nothing -- silently, with no error. The global-variable attempt failed
# for a different reason (see below), which is why swapping between the
# two never helped.
#
# Every bulk action below now finds the player the same way the game-start
# on_action does -- `every_country = { limit = { is_player = yes }
# save_scope_as = ... }` -- which is proven to work, and depends on
# neither `root` nor a stored pointer.
#
# UX (per the user, 2026-09-07): "if I click select all or deselect all in
# a tab, all the countries in that tab get selected/deselected", full
# stop. So Deselect All clears ALL FOUR flags for everything matching the
# tab's own filter, not just that category's flag.

watchlist_bulk_select_great_power_sgui = {
	scope = country

	effect = {
		# Establish the player WITHOUT relying on `root`. Confirmed by the
		# 2026-09-07 root probes: all three SetRoot forms do execute the
		# effect, but `[THIS.GetCountry...]` never resolved, i.e. the root
		# handed to a button-triggered scripted GUI is not a usable
		# country. This mirrors the pattern that provably works in
		# common/on_actions/00_smart_notifications_on_actions.txt's
		# game-start hook instead: find the player by trigger, save it,
		# then compare against the saved scope.
		every_country = {
			limit = { is_player = yes }
			save_scope_as = snw_bulk_player
		}

		every_country = {
			limit = {
				NOT = { this = scope:snw_bulk_player }
				NOT = { is_country_type = decentralized }
				country_rank >= rank_value:great_power
			}
			set_variable = watched_via_great_power
		}
	}

	ai_is_valid = { always = no }
	ai_chance = { base = 0 }
}
