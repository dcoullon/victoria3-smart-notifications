# Smart Notifications mod: Country Watchlist (Phase 3, see TODO.md for the
# full spec). Toggles whether a country is on the player's watchlist.
#
# Data model (TODO.md "Data model — REVISED 2026-09-06"): 4 separate
# provenance flags (great_power/neighbor/rival/manually), not 1, so a bulk
# category action can't silently un-watch a country kept for a different
# reason. This SGUI is the INDIVIDUAL toggle (the country-panel star and,
# later, a row checkbox on the management screen) -- per the spec, that
# means "watched or not, period": clears all four flags to remove, or sets
# just `watched_manually` to add (a star click doesn't know about
# category membership, so a fresh watch via this button is always
# recorded as a manual one).
#
# Structure confirmed against `common/scripted_guis/scripted_guis.md` and
# real working examples (`debug_sguis.txt`'s single-scope toggles).
# CORRECTED 2026-09-07, real bug found live (confirmed by the user: the
# checkbox never toggled, even after a full window close/reopen -- so it
# wasn't a redraw/caching issue, the click was being refused outright).
# Root cause: `is_valid` is not just a display query, it's the actual gate
# that determines whether `.Execute()` is allowed to run at all ("Determines
# whether the SGUI can be used by a player" per scripted_guis.md). Setting
# it to "is this country already watched" meant clicking to WATCH an
# unwatched country was always refused, since is_valid was false for
# exactly the countries you'd want to click.
#
# Fixed by splitting into two SGUIs, the same way vanilla does when a check
# and an action need different validity rules -- confirmed via a real
# working example, `je_meiji_restoration_japanese_emperor_check_sgui`
# (journal_entry_sguis.txt), which is `is_valid`-only with NO effect at
# all, used purely for a `.IsValid()` display check elsewhere.
watchlist_toggle_sgui = {
	scope = country

	# No is_valid here on purpose -- toggling is always a valid action
	# regardless of current watched state, on or off.

	effect = {
		if = {
			limit = {
				OR = {
					has_variable = watched_via_great_power
					has_variable = watched_via_neighbor
					has_variable = watched_via_rival
					has_variable = watched_manually
				}
			}
			remove_variable = watched_via_great_power
			remove_variable = watched_via_neighbor
			remove_variable = watched_via_rival
			remove_variable = watched_manually
		}
		else = {
			set_variable = watched_manually
		}
	}

	# Per CLAUDE.md: scripted GUIs must never be available to the AI.
	ai_is_valid = { always = no }
	ai_chance = { base = 0 }
}

# Read-only companion to watchlist_toggle_sgui above -- ONLY used via
# `.IsValid(...)` for a checkbox's `checked` state in
# gui/message_settings.gui, never `.Execute()`'d (no effect defined at
# all, matching the je_meiji_restoration_japanese_emperor_check_sgui
# precedent this pattern is based on).
watchlist_is_watched_check_sgui = {
	scope = country

	is_valid = {
		OR = {
			has_variable = watched_via_great_power
			has_variable = watched_via_neighbor
			has_variable = watched_via_rival
			has_variable = watched_manually
		}
	}

	ai_is_valid = { always = no }
	ai_chance = { base = 0 }
}

# --- Category-tab row filters (Phase 3 categories, added 2026-09-07) ---
# Each is read-only (is_valid-only, no effect), same shape as
# watchlist_is_watched_check_sgui above, used purely to decide whether a
# given row is visible on a given category sub-tab in
# gui/message_settings.gui.
#
# Great Powers is a TRUE live check (country_rank is absolute, no
# comparison to the player needed) -- browsing shows anyone CURRENTLY a
# Great Power, watched or not, exactly per the original spec.
#
# Neighbors and Rivals -- REVISED 2026-09-07, now TRUE live checks too.
# The engine-notes.md blocker ("no confirmed way to reference the player
# as a GuiScope root/AddScope value") turned out to only block one
# specific technique. Found a different, fully-confirmed way that sidesteps
# it entirely: `any_country = { is_player = yes <triggers using root> }` is
# real, working vanilla syntax (confirmed: common/scripted_buttons/
# 00_balkan_wars_buttons.txt, events/ethiopia.txt, events/krakow_events.txt
# all use exactly this "find the player from inside a trigger block"
# idiom) -- `root` stays bound to the scripted_gui's own invocation root
# (the row's country) throughout the nesting, so this compares the ROW's
# country against the player without ever needing to pass the player IN
# as a scope from the GUI side. No more guessing needed here.
watchlist_is_great_power_check_sgui = {
	scope = country

	is_valid = {
		country_rank >= rank_value:great_power
	}

	ai_is_valid = { always = no }
	ai_chance = { base = 0 }
}

watchlist_is_neighbor_check_sgui = {
	scope = country

	is_valid = {
		# Is the player adjacent to this row's country?
		any_country = {
			is_player = yes
			is_adjacent_to_country = root
		}
	}

	ai_is_valid = { always = no }
	ai_chance = { base = 0 }
}

watchlist_is_rival_check_sgui = {
	scope = country

	is_valid = {
		# Does the player have this row's country as a declared rival?
		# any_rival_country ("countries being rivaled by the country in
		# scope") from the player's scope, comparing each candidate back
		# to root -- the `?= ` scope-equality comparator confirmed via
		# common/power_bloc_names/00_power_bloc_names.txt ("c:FRA ?= this").
		any_country = {
			is_player = yes
			any_rival_country = {
				this ?= root
			}
		}
	}

	ai_is_valid = { always = no }
	ai_chance = { base = 0 }
}

watchlist_is_flagged_manual_check_sgui = {
	scope = country

	is_valid = {
		has_variable = watched_manually
	}

	ai_is_valid = { always = no }
	ai_chance = { base = 0 }
}

# --- Category bulk select/deselect (added 2026-09-07) ---
# TODO.md flagged this as blocked on "no confirmed way to reference the
# player as a GuiScope root from a plain button click (not tied to any
# row)". Resolved the same way as the live Neighbors/Rivals checks above,
# but via a DIFFERENT confirmed technique that fits an effect block
# better: bind the button's own container to
# `datacontext = "[GetMetaPlayer.GetPlayedOrObservedCountry]"` in
# gui/message_settings.gui (confirmed-real accessor -- used identically
# in vanilla's market_panel.gui, right_click_menu.gui, ingame_hud.gui to
# set a widget's Country context to the player without any row/list
# involved), so `Country.MakeScope` inside that container IS the player.
# `root` in every effect below is therefore already the player -- exactly
# mirroring the proven pattern in
# common/on_actions/00_smart_notifications_on_actions.txt's game-start
# bulk-select (same `every_rival_country` / `is_adjacent_to_country = root`
# / `country_rank >= rank_value:great_power` shapes, just re-triggerable
# on demand instead of once at campaign start).
#
# Deselect only removes THAT category's own flag, never the others --
# per the multi-flag data model (TODO.md), a country also watched via a
# different provenance correctly stays on the effective watchlist.
watchlist_bulk_select_great_power_sgui = {
	scope = country

	effect = {
		every_country = {
			limit = {
				NOT = { this = root }
				country_rank >= rank_value:great_power
			}
			set_variable = watched_via_great_power
		}
	}

	ai_is_valid = { always = no }
	ai_chance = { base = 0 }
}

watchlist_bulk_deselect_great_power_sgui = {
	scope = country

	effect = {
		every_country = {
			limit = { has_variable = watched_via_great_power }
			remove_variable = watched_via_great_power
		}
	}

	ai_is_valid = { always = no }
	ai_chance = { base = 0 }
}
