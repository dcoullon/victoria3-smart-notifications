# Smart Notifications mod: "notify me when this specific law has more
# support than pushback" opt-in checkbox (see TODO.md's sized backlog,
# item 4, and common/alert_types/01_smart_notifications_alerts.txt's
# smart_notifications_law_commitment_alert). GUI anchor:
# gui/politics_panel_change_law.gui's law detail panel (shown for whichever
# law the player has selected or is enacting).
#
# REVISED 2026-09-08 per the user: this must flag a SPECIFIC law, not "the
# enactment currently in progress" (the first version built here) -- the
# whole point is finding out a law is now worth enacting BEFORE committing
# to it, not just tracking one you already started.
#
# Root = `law` scope (scope = law, NOT country and NOT law_type). `law` is
# a country's own per-law-type record -- confirmed real and fully
# scriptable via `any_law`/`any_active_law` (triggers.log: "Supported
# Targets: law") and already navigable via effects/triggers in this exact
# mod (common/alert_types/01_smart_notifications_alerts.txt's amendment
# alert already does `any_active_law = { any_scope_amendment = { ... } }`).
# This is deliberately NOT `law_type` (the global, country-agnostic
# database entry for a law) -- flagging on `law_type` would make "wanted"
# a single global setting shared by every save/country, when what's needed
# is "wanted BY THIS COUNTRY, in THIS campaign," which a `law` record
# (already inherently per-country) gives for free.
#
# UNCONFIRMED, the single biggest risk in this whole feature: `Law.MakeScope`
# in the GUI file. `.MakeScope` is confirmed to exist on several other GUI
# datacontext object types with a matching real scriptable scope --
# Character, State, Country, PoliticalMovement (all used this way
# elsewhere in vanilla) -- and `Law` here is exactly analogous (a GUI
# object wrapping a real, documented `law` scope), but no vanilla file
# anywhere actually calls `Law.MakeScope`, so this is extrapolation from a
# consistent pattern, not a confirmed precedent. First thing to check
# in-game if the checkbox doesn't render or doesn't toggle.
#
# Split into toggle/check pairs for the same reason as
# common/scripted_guis/watchlist_sgui.txt: `is_valid` also gates whether
# `.Execute()` is allowed to run at all, so it can't double as the toggle's
# own "am I currently set" query.
#
# TEMPORARY instrumentation (2026-09-08): the SNW_LAW_NOTIFY debug_log below
# exists specifically to test the Law.MakeScope risk above. If the checkbox
# click produces no visible change AND no SNW_LAW_NOTIFY line in debug.log,
# the effect never ran at all -- Law.MakeScope (or the GetScriptedGui/.Execute
# call built on it) failed silently. If the line DOES appear but the checkbox
# still doesn't visibly toggle, the write succeeded and the bug is instead in
# the `checked =` read-back (still Law.MakeScope, same call) or in the
# checkbox widget itself. Delete once confirmed working.
smart_notifications_law_notify_toggle_sgui = {
	scope = law

	# No is_valid -- toggling is always a valid action regardless of
	# current state, matching watchlist_toggle_sgui's own reasoning.
	effect = {
		if = {
			limit = {
				has_variable = smart_notifications_wanted_law
			}
			remove_variable = smart_notifications_wanted_law
			debug_log = "SNW_LAW_NOTIFY|toggle|law=[THIS.GetNameNoFormatting]|country=[THIS.GetCountry.GetNameNoFormatting]|UNFLAGGED"
		}
		else = {
			set_variable = smart_notifications_wanted_law
			debug_log = "SNW_LAW_NOTIFY|toggle|law=[THIS.GetNameNoFormatting]|country=[THIS.GetCountry.GetNameNoFormatting]|FLAGGED"
		}
	}

	# Per CLAUDE.md: scripted GUIs must never be available to the AI.
	ai_is_valid = { always = no }
	ai_chance = { base = 0 }
}

smart_notifications_law_notify_check_sgui = {
	scope = law

	is_valid = {
		has_variable = smart_notifications_wanted_law
	}

	ai_is_valid = { always = no }
	ai_chance = { base = 0 }
}
