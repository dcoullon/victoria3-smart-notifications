# Smart Notifications mod: one-time toast per state that enters a
# taxation deficit, independent of the persistent
# smart_notifications_taxation_deficit_alert's dismiss state (see
# common/alert_types/01_smart_notifications_alerts.txt).
#
# ADDED 2026-09-09 per the user, after confirming the exact same
# dismiss-hides-new-instances gap the law commitment toast was built for
# also applies here: the alert is grouped/aggregated
# (alert_group = smart_notifications_taxation_deficit_states) and its
# dismissal is edge-triggered on the alert's OWN valid (any_scope_state
# in deficit) going false->true -- if State A's deficit already showed
# and got dismissed, and a DIFFERENT state B later develops a deficit
# while A's is still ongoing, "some state has a deficit" never flips
# false->true again, so the alert never resurfaces for the new state B.
#
# States are tracked individually via a country-scope VARIABLE LIST
# holding event-target references to the actual state objects
# (add_to_variable_list / is_target_in_variable_list -- confirmed real
# via script_docs effects.log/triggers.log), so one shared message
# definition suffices, exactly like vanilla's own `type = state` messages
# (e.g. colony_created) do -- confirmed via
# localization/english/interfaces_l_english.yml's own
# `[SCOPE.GetRootScope.GetState.GetName]` usage in exactly that message's
# loc, the SAME cast chain our own taxation_deficit_alert's loc already
# uses successfully (confirmed working live).
#
# CAP -- REDESIGNED 2026-09-09, THIRD ITERATION, deliberately simpler
# than the first two: 3 toasts TOTAL PER CAMPAIGN, not 3 concurrent
# slots that free up as states recover. The first two attempts (a
# same-pass mutation-visibility bug, then a wrong-scope membership
# check) were both bugs in the "3 concurrent, refillable" tracking
# logic specifically -- and even once that logic was correctly fixed,
# the user reported the RESULT still wasn't what they wanted: it kept
# cycling through a stream of different states over time as old ones
# recovered and new slots opened, which still reads as noisy, just with
# rotating targets instead of a stuck one. Per the user's own call
# ("last attempt... otherwise just the alert, no toast"), this drops the
# "recover and re-open a slot" mechanism ENTIRELY rather than trying to
# retune it a third time -- a state added to the notified list here
# NEVER leaves it, so once 3 states have ever triggered a toast this
# campaign, the door is permanently shut. This is a real behavior
# change, not just a bug fix: an isolated, meaningful deficit occurring
# long after the campaign's first 3 will now stay silent (the persistent
# ALERT, untouched by any of this, still shows it) -- accepted as the
# right trade-off specifically because it removes an entire class of
# state-tracking complexity that has been the source of every bug in
# this file so far, in favor of a monotonic counter that cannot drift.
#
# `ordered_scope_state` (confirmed real, script_docs effects.log;
# confirmed to actually run its effect body against the WHOLE selected
# batch, not just one entity, via common/on_actions/00_on_actions_monthly.txt's
# exile-pool culling on_action, which uses the identical
# ordered_X + computed max + single effect-line shape) still does the
# actual selection, `max = 3` per call. `order_by` uses the same
# nested-block literal as before (`order_by = { value = 1 }`, confirmed
# real shape from that same exile-pool example) since there's still no
# real ranking criterion available (tax_capacity/usage are
# comparison-only, not extractable as a number).
#
# The OUTER `variable_list_size >= 3` gate is still required and kept --
# `max = 3` only limits how many get selected IN ONE MONTHLY CALL, it is
# NOT a running total across separate calls. Without the outer gate, a
# genuinely new 4th state discovered a year later would still be
# eligible (it was never a list member) and would still fire, since
# nothing else would stop it. The gate is what makes 3 a true, permanent,
# whole-campaign ceiling: once the list reaches 3 members and nothing
# ever removes any (no reset branch exists anymore), the gate blocks
# every future pulse's selection outright, regardless of how many new
# states become eligible afterward.
#
# The membership check bug found live 2026-09-09 (THIS inside
# ordered_scope_state's own `limit` is the candidate state, not the
# country -- the list lives on the country, so the check must re-scope
# via `THIS.owner` first) is preserved here; that part was correct, only
# the "does membership ever get removed" design was wrong.
on_monthly_pulse_country = {
	on_actions = {
		smart_notifications_taxation_deficit_toast_check
	}
}

smart_notifications_taxation_deficit_toast_check = {
	effect = {
		if = {
			limit = {
				is_player = yes
				NOT = {
					variable_list_size = {
						name = smart_notifications_notified_deficit_states
						target >= 3
					}
				}
			}
			ordered_scope_state = {
				limit = {
					is_incorporated = yes
					tax_capacity < tax_capacity_usage
					NOT = {
						THIS.owner = {
							is_target_in_variable_list = {
								name = smart_notifications_notified_deficit_states
								target = prev
							}
						}
					}
				}
				order_by = { value = 1 }
				max = 3
				check_range_bounds = no
				THIS.owner = {
					add_to_variable_list = {
						name = smart_notifications_notified_deficit_states
						target = prev
					}
					prev = {
						post_notification = smart_notifications_taxation_deficit_toast
					}
					debug_log = "SNW_TAX_TOAST|fired|state=[prev.GetState.GetName]|date=[TimeKeeper.GetCurrentDate.GetString]"
				}
			}
		}
	}
}
