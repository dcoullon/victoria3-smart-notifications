# Smart Notifications mod: one-time toast per state that enters a
# taxation deficit, independent of the persistent
# smart_notifications_taxation_deficit_alert's dismiss state (see
# common/alert_types/01_smart_notifications_alerts.txt).
#
# ADDED 2026-09-09 per the user, after confirming the exact same
# dismiss-hides-new-instances gap the law commitment toast was built for
# also applies here: the alert is grouped/aggregated
# (alert_group = smart_notifications_taxation_deficit_states) and its
# dismissal is edge-triggered on the alert's OWN valid (any_scope_state
# in deficit) going false->true -- if State A's deficit already showed
# and got dismissed, and a DIFFERENT state B later develops a deficit
# while A's is still ongoing, "some state has a deficit" never flips
# false->true again, so the alert never resurfaces for the new state B.
#
# UNLIKE the law toast, this does NOT need a per-type static dispatch --
# there is no small, fixed, enumerable set of "state types" the way there
# are exactly 138 law types (see
# common/on_actions/09_smart_notifications_law_ready_toast.txt's own
# header for why THAT one needs 138 branches). States are tracked
# individually via a country-scope VARIABLE LIST holding event-target
# references to the actual state objects (add_to_variable_list /
# is_target_in_variable_list / remove_list_variable -- all confirmed real
# via script_docs effects.log/triggers.log), so one shared message
# definition suffices, exactly like vanilla's own `type = state` messages
# (e.g. colony_created) do -- confirmed via
# localization/english/interfaces_l_english.yml's own
# `[SCOPE.GetRootScope.GetState.GetName]` usage in exactly that message's
# loc, the SAME cast chain our own taxation_deficit_alert's loc already
# uses successfully (confirmed working live this session).
#
# CAP, FIRST ATTEMPT 2026-09-09 (WRONG, replaced below): tried gating the
# fire condition on `NOT = { variable_list_size >= 3 }` INSIDE the same
# `every_scope_state` loop that also does `add_to_variable_list` for each
# match. CONFIRMED BROKEN live: the user still got 30 toasts in one pulse
# (debug.log showed 30 consecutive `SNW_TAX_TOAST|fired` lines, all same
# timestamp). Root cause: `every_scope_state`'s `limit` for each state
# does not see the `add_to_variable_list` mutations made by EARLIER
# states in the SAME pass -- every state's limit is evaluated against
# the list's state at the START of this pulse, not updated live as the
# loop runs. This is a real, confirmed engine behavior difference from
# what was assumed, not a guess corrected in the abstract -- the fix
# below is a structurally different mechanism specifically BECAUSE this
# one failed, not a tweak of the same approach.
#
# CAP, ACTUAL FIX: `ordered_scope_state` (confirmed real,
# script_docs effects.log, same iterator family used by vanilla's own
# common/scripted_effects/00_victoria_ip4_scripted_effects.txt for
# picking states) has SELECT-THEN-ACT semantics BY DESIGN -- it picks its
# `max`-capped subset of matching states FIRST (limit-filtered, before
# any effect in its body runs for ANY of them), then runs the effect body
# for exactly that subset. This sidesteps the specific mid-loop-mutation-
# visibility bug above entirely: `max` is an engine-enforced cap on the
# selection itself, not something computed by watching side effects
# accumulate turn-by-turn inside the loop.
#
# `order_by` has no meaningful criterion available here (tax_capacity/
# tax_capacity_usage are confirmed comparison-only, not extractable as a
# raw number -- same limitation as the law alert's success/stall values,
# see docs/engine-notes.md "Success-vs-stall comparisons have no numeric
# form"). Uses a flat literal (order_by = 1, every candidate tied) since
# WHICH 3 states get picked doesn't matter, only that no more than 3 do.
#
# KNOWN IMPRECISION, disclosed rather than hidden: this checks the
# currently-active count ONCE per pulse (`NOT variable_list_size >= 3`,
# gating the whole ordered_scope_state block) and then lets
# `max = 3` add up to 3 MORE regardless of that starting count -- so if
# 1 or 2 states were already active going into a pulse, this pulse could
# push the total to 4 or 5, not hold it at a strict 3. This is NOT the
# reported problem (many states going into deficit simultaneously,
# starting from 0 active) -- for that case (0 already active, gate
# passes, up to 3 new ones fire) this is exactly correct. A fully precise
# cap would need `max` to be a computed `3 - <current count>` script
# value; not attempted here since no confirmed-real vanilla precedent for
# reading variable_list_size as a raw number (as opposed to a
# >=/<=-comparison trigger) was found, and guessing at that specific
# syntax is exactly the mistake being corrected in this same file already.
on_monthly_pulse_country = {
	on_actions = {
		smart_notifications_taxation_deficit_toast_check
	}
}

smart_notifications_taxation_deficit_toast_check = {
	effect = {
		if = {
			limit = {
				is_player = yes
				NOT = {
					variable_list_size = {
						name = smart_notifications_notified_deficit_states
						target >= 3
					}
				}
			}
			ordered_scope_state = {
				limit = {
					is_incorporated = yes
					tax_capacity < tax_capacity_usage
					NOT = {
						is_target_in_variable_list = {
							name = smart_notifications_notified_deficit_states
							target = THIS
						}
					}
				}
				order_by = 1
				max = 3
				check_range_bounds = no
				THIS.owner = {
					add_to_variable_list = {
						name = smart_notifications_notified_deficit_states
						target = prev
					}
					prev = {
						post_notification = smart_notifications_taxation_deficit_toast
					}
					debug_log = "SNW_TAX_TOAST|fired"
				}
			}
		}
		if = {
			limit = { is_player = yes }
			every_scope_state = {
				limit = {
					NOT = {
						AND = {
							is_incorporated = yes
							tax_capacity < tax_capacity_usage
						}
					}
				}
				THIS.owner = {
					if = {
						limit = {
							is_target_in_variable_list = {
								name = smart_notifications_notified_deficit_states
								target = prev
							}
						}
						remove_list_variable = {
							name = smart_notifications_notified_deficit_states
							target = prev
						}
						debug_log = "SNW_TAX_TOAST|reset"
					}
				}
			}
		}
	}
}
