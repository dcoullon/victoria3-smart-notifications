# Smart Notifications mod: "notify me when this law looks likely to pass"
# (see TODO.md's sized backlog item 4,
# common/scripted_guis/smart_notifications_law_notify_sgui.txt for the
# checkbox toggle, and
# common/alert_types/01_smart_notifications_alerts.txt's
# smart_notifications_law_commitment_alert for the actual notification).
#
# This file's only job: clear a stale `smart_notifications_watching_enactment`
# flag whenever a NEW enactment starts. The flag means "notify me about the
# enactment currently in progress," not "notify me about a specific law
# type" (see the scripted_gui file's header comment for why), so if the
# player checked the box, then cancelled and started enacting a DIFFERENT
# law before the first one resolved, the flag would otherwise carry over
# onto an enactment they never actually opted into.
#
# `on_law_enactment_started` confirmed real via vanilla's own
# common/on_actions/00_code_on_actions.txt (already fires vanilla's own
# Montenegro/monarchy-abolition script there); root = country, confirmed by
# vanilla's own unscoped `is_enacting_law = law_type:X` calls in the same
# block.
on_law_enactment_started = {
	on_actions = {
		smart_notifications_law_commitment_reset
	}
}

smart_notifications_law_commitment_reset = {
	effect = {
		remove_variable = smart_notifications_watching_enactment
	}
}
