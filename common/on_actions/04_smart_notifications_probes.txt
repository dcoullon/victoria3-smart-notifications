# TEMPORARY diagnostic probes -- DELETE once the third-party notification
# filtering is scoped (Phase 4, see TODO.md). Adds no player-visible
# behaviour: every block below only writes debug_log lines.
#
# Purpose: the third-party ("something happened between two other
# countries") notifications are the main remaining filtering target, but
# several of them are rooted on object types we have no confirmed way to
# reach the involved countries from -- `# Root = Diplomatic Action` and
# `# Root = Diplomatic Pact` per vanilla's own comments in
# common/on_actions/00_code_on_actions.txt. The script_docs event_targets
# dump documents no scope links out of a diplomatic_action at all, so the
# only way to learn what is actually bound is to observe a real one.
#
# We already have a confirmed error from guessing here (`is_player trigger
# [ Wrong scope for trigger: diplomatic_action, expected country ]`), so
# every probe below is written defensively:
#   - `scope:x ?= { ... }` -- the `?=` guard means a scope that does not
#     exist is skipped rather than erroring (confirmed vanilla pattern,
#     e.g. `scope:target ?= { ... }` in vanilla's own on_war_end).
#   - the name is read as `[THIS.GetCountry.GetNameNoFormatting]` from INSIDE that
#     scope, so we never apply a country accessor to a non-country root.
#   - no `is_player` anywhere near a non-country root.
#
# After a playtest: grep debug.log for SNW_PROBE to see which on_actions
# fired, which scope names exist on each, and which countries they name.

# ADDED 2026-09-09 -- see 06_smart_notifications_diplomatic_action_filtering.txt's
# header comment and TODO.md's "BUG: routine-relations demotion never
# actually fires" entry for the full story. `has_diplomatic_pact = {
# who = X type = increase_relations/damage_relations }` (checked from
# scope:actor, both with and without is_initiator) has returned "no" in
# EVERY real firing so far (Siam, Bulungan -> Great Qing), even though
# these were genuinely relations-change actions. Two live theories:
# (a) the pact isn't registered in gamestate yet at the EXACT instant
# on_diplomatic_action fires (timing), or (b) these one-sided actions
# never create a `has_diplomatic_pact`-queryable pact object at all, ever
# (the user's own hypothesis). This sweep runs monthly from the player's
# own scope and checks EVERY other country for an existing
# increase_relations/damage_relations pact with the player, regardless
# of when it started -- if Siam or Bulungan (or any other country that
# already fired this notification) EVER shows up here on a later pulse,
# that confirms (a) (pure timing gap) and rules out (b). If NEITHER
# ever shows up despite the player clearly having had such actions taken
# on them, that's strong evidence for (b) instead. Delete once the real
# cause is found. Grep debug.log for SNW_PACT_SWEEP.
#
# FIXED before its first live run (2026-09-09): the version first
# committed would have logged NOTHING for the per-country sweep whatever
# the answer was -- its debug_log used bare `[THIS.GetNameNoFormatting]`,
# the exact uncast form docs/engine-notes.md's own "THIS needs a cast"
# section records as always failing ("Data error in loc string"), and
# error.log confirms it still failing right now for SNW_LAW_TOAST's
# identical bare call. A silent zero would have read as evidence for (b)
# when it was really just a blind tap (see the same section's "when a
# debug tap prints nothing useful, check the tap itself"). Cast added.
# `[TimeKeeper.GetCurrentDate.GetString]` was checked the same way and IS
# real (vanilla interfaces_l_english.yml CLOCK_TEXT, and event flavor
# text in ip2_misc_01_l_english.yml -- both genuine dynamic-text
# contexts, not .gui bindings), so it stays.
#
# Instrument 1 below was added at the same time as a second, independent
# reading: `every_scope_diplomatic_pact` + `is_diplomatic_action_type`
# (both confirmed in effects.log/triggers.log for country and
# diplomatic_pact scope respectively) enumerate the player's actual pact
# OBJECTS rather than querying for one. If hypothesis (b) is right, no
# relations pact object exists to enumerate either, and both instruments
# stay silent together -- agreement between two unrelated mechanisms is
# what makes a null result trustworthy here.
on_monthly_pulse_country = {
	on_actions = { smart_notifications_probe_pact_sweep }
}

smart_notifications_probe_pact_sweep = {
	effect = {
		if = {
			limit = { is_player = yes }
			debug_log = "SNW_PACT_SWEEP|sweep_ran|date=[TimeKeeper.GetCurrentDate.GetString]"

			# Instrument 1 -- enumerate the pact OBJECTS the player is a
			# party to, straight from gamestate, and ask each one its
			# action type (`is_diplomatic_action_type`, confirmed real on
			# `diplomatic_pact` scope via triggers.log). Independent of
			# `has_diplomatic_pact` entirely: if a one-sided relations
			# pact object exists at all, it shows up here even if the
			# has_diplomatic_pact query below somehow misses it.
			every_scope_diplomatic_pact = {
				debug_log = "SNW_PACT_SWEEP|player_pact_seen"
				if = {
					limit = { is_diplomatic_action_type = increase_relations }
					debug_log = "SNW_PACT_SWEEP|player_pact|type=increase_relations"
				}
				if = {
					limit = { is_diplomatic_action_type = damage_relations }
					debug_log = "SNW_PACT_SWEEP|player_pact|type=damage_relations"
				}
			}

			# Instrument 2 -- the original per-country has_diplomatic_pact
			# sweep, i.e. exactly the query the demotion feature uses,
			# just run later in time instead of at the on_action instant.
			every_country = {
				limit = {
					NOT = { is_player = yes }
				}
				if = {
					limit = { has_diplomatic_pact = { who = ROOT type = increase_relations } }
					debug_log = "SNW_PACT_SWEEP|country=[THIS.GetCountry.GetNameNoFormatting]|type=increase_relations|is_initiator=unspecified|result=yes"
				}
				if = {
					limit = { has_diplomatic_pact = { who = ROOT type = damage_relations } }
					debug_log = "SNW_PACT_SWEEP|country=[THIS.GetCountry.GetNameNoFormatting]|type=damage_relations|is_initiator=unspecified|result=yes"
				}
				if = {
					limit = { has_diplomatic_pact = { who = ROOT type = increase_relations is_initiator = yes } }
					debug_log = "SNW_PACT_SWEEP|country=[THIS.GetCountry.GetNameNoFormatting]|type=increase_relations|is_initiator=yes|result=yes"
				}
				if = {
					limit = { has_diplomatic_pact = { who = ROOT type = increase_relations is_initiator = no } }
					debug_log = "SNW_PACT_SWEEP|country=[THIS.GetCountry.GetNameNoFormatting]|type=increase_relations|is_initiator=no|result=yes"
				}
			}
		}
	}
}
