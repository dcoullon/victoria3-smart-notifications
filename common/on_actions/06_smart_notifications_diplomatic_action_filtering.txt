# F1 of docs/watchlist-spec.md -- generic diplomatic actions, routed into
# five cells by who the action is aimed at and who took it.
#
# Root = Diplomatic Action (`on_diplomatic_action`, confirmed via vanilla's
# own comment in common/on_actions/00_code_on_actions.txt). scope:actor and
# scope:recipient are confirmed real and bound here -- unlike the sibling
# `_third_party_` on_actions, which stay blocked: a real error
# (`is_player trigger [ Wrong scope for trigger: diplomatic_action,
# expected country ]`) already came from guessing at those.
#
# This file was rewritten on 2026-09-09 against the locked spec, replacing
# a day's worth of accumulated ad-hoc rules. The history of how those rules
# went wrong -- the double-notification hunt, the raw-loc-key bug, three
# separate failed attempts to read the action's TYPE -- is in TODO.md and
# docs/engine-notes.md, deliberately not repeated here. What matters for
# reading the code below:
#
# 1. The action type is unknowable at this moment. Not "hard": three
#    mechanisms were tried and all three are closed (engine-notes.md
#    § The pact for a diplomatic action does not exist yet, § A .md schema
#    doc is not proof a key parses). Do not add a fourth attempt without
#    genuinely new information.
# 2. The rule therefore keys off WHO, not WHAT. Routine relations traffic
#    is demoted by proxy -- an unwatched actor -- rather than by name.
# 3. Vanilla's per-action `_desc` text assumes the reader is a party to the
#    action, so only the at_player_* keys may reuse it. The other three use
#    generic wording. See common/messages/00_messages.txt.
# 4. `smart_notifications_is_watched` counts the player as watched, so
#    is_player is always tested first, and actor-side tests use
#    smart_notifications_is_watched_not_player.

on_diplomatic_action = {
	on_actions = {
		smart_notifications_diplomatic_action
	}
}

# Root = Diplomatic Action. OR'd across actor and recipient (unlike the
# diplo-play events' any_scope_play_involved, there's no roster to iterate
# here -- just the two countries directly involved) so the notification
# elevates if EITHER side is watched or is the player -- matches the
# established "notifications about the player must never be filtered"
# rule automatically, since is_player = yes is already baked into
# smart_notifications_is_watched (00_smart_notifications_triggers.txt): a
# diplomatic action directed at the player (scope:recipient = us) always
# elevates regardless of watchlist.
smart_notifications_diplomatic_action = {
	effect = {
		# F1 of docs/watchlist-spec.md, implemented exactly. Two questions,
		# asked in this order:
		#
		#   1. WHO is it aimed at?  you > a watched country > nobody
		#      you care about
		#   2. WHO did it?          a watched country other than you, or
		#      anyone else
		#
		# and the five combinations map one-to-one onto five message keys,
		# each in its own group so the player can retune any single cell
		# from Message Settings (spec D11).
		#
		# The action TYPE never appears, because it cannot: at the moment
		# this on_action runs there is no way to tell Increase Relations
		# from a rivalry declaration. Three mechanisms were tried and all
		# three are closed -- see docs/engine-notes.md. That is why the
		# rule keys off who acted rather than what they did, and why the
		# routine relations traffic is demoted by proxy (an unwatched
		# actor) rather than by name.
		#
		# `is_player` is checked BEFORE `smart_notifications_is_watched`
		# throughout, because the latter deliberately returns yes for the
		# player (00_smart_notifications_triggers.txt) -- without that
		# ordering every player-aimed action would fall into the watched
		# branch instead of its own.
		#
		# Actor-side tests use smart_notifications_is_watched_not_player,
		# never the plain trigger: spec invariant 5, your own actions are
		# never news. Without it, improving relations with a watched
		# country would toast your own diplomacy back at you.
		if = {
			limit = { scope:recipient ?= { is_player = yes } }

			if = {
				limit = { scope:actor ?= { smart_notifications_is_watched_not_player = yes } }
				post_notification = smart_notifications_diplomatic_action_at_player_by_watched
				debug_log = "SNW_FILTER|diplomatic_action|posted_key=smart_notifications_diplomatic_action_at_player_by_watched|actor=[SCOPE.sC('actor').GetNameNoFormatting]|recipient=[SCOPE.sC('recipient').GetNameNoFormatting]|AT_PLAYER_BY_WATCHED"
			}
			else = {
				post_notification = smart_notifications_diplomatic_action_at_player_by_other
				debug_log = "SNW_FILTER|diplomatic_action|posted_key=smart_notifications_diplomatic_action_at_player_by_other|actor=[SCOPE.sC('actor').GetNameNoFormatting]|recipient=[SCOPE.sC('recipient').GetNameNoFormatting]|AT_PLAYER_BY_OTHER"
			}
		}
		else_if = {
			# Aimed AT a watched country. Deliberately does not test the
			# actor here -- a watched country acting on a stranger is not
			# an event aimed at anything you care about, and falls through
			# to the `elsewhere` branch below. That distinction is the
			# whole point of spec definition "aimed at", and getting it
			# wrong is what produced the 13-toast spam report of v0.37.
			limit = { scope:recipient ?= { smart_notifications_is_watched = yes } }

			if = {
				limit = { scope:actor ?= { smart_notifications_is_watched_not_player = yes } }
				post_notification = smart_notifications_diplomatic_action_at_watched_by_watched
				debug_log = "SNW_FILTER|diplomatic_action|posted_key=smart_notifications_diplomatic_action_at_watched_by_watched|actor=[SCOPE.sC('actor').GetNameNoFormatting]|recipient=[SCOPE.sC('recipient').GetNameNoFormatting]|AT_WATCHED_BY_WATCHED"
			}
			else = {
				post_notification = smart_notifications_diplomatic_action_at_watched_by_other
				debug_log = "SNW_FILTER|diplomatic_action|posted_key=smart_notifications_diplomatic_action_at_watched_by_other|actor=[SCOPE.sC('actor').GetNameNoFormatting]|recipient=[SCOPE.sC('recipient').GetNameNoFormatting]|AT_WATCHED_BY_OTHER"
			}
		}
		else = {
			post_notification = smart_notifications_diplomatic_action_elsewhere
			debug_log = "SNW_FILTER|diplomatic_action|posted_key=smart_notifications_diplomatic_action_elsewhere|actor=[SCOPE.sC('actor').GetNameNoFormatting]|recipient=[SCOPE.sC('recipient').GetNameNoFormatting]|ELSEWHERE"
		}

		# Per-side watched status, logged alongside every decision. This is
		# how the matrix gets verified without the user watching for
		# toasts: each event prints its own cell plus the two inputs that
		# chose it, so a whole session's routing can be checked from
		# debug.log afterwards. Keep until the matrix is confirmed live.
		scope:actor ?= {
			if = {
				limit = { smart_notifications_is_watched_not_player = yes }
				debug_log = "SNW_FILTER|diplomatic_action|involved|country=[THIS.GetCountry.GetNameNoFormatting]|role=actor|watched=yes"
			}
			else_if = {
				limit = { is_player = yes }
				debug_log = "SNW_FILTER|diplomatic_action|involved|country=[THIS.GetCountry.GetNameNoFormatting]|role=actor|watched=self"
			}
			else = {
				debug_log = "SNW_FILTER|diplomatic_action|involved|country=[THIS.GetCountry.GetNameNoFormatting]|role=actor|watched=no"
			}
		}
		scope:recipient ?= {
			if = {
				limit = { is_player = yes }
				debug_log = "SNW_FILTER|diplomatic_action|involved|country=[THIS.GetCountry.GetNameNoFormatting]|role=recipient|watched=self"
			}
			else_if = {
				limit = { smart_notifications_is_watched = yes }
				debug_log = "SNW_FILTER|diplomatic_action|involved|country=[THIS.GetCountry.GetNameNoFormatting]|role=recipient|watched=yes"
			}
			else = {
				debug_log = "SNW_FILTER|diplomatic_action|involved|country=[THIS.GetCountry.GetNameNoFormatting]|role=recipient|watched=no"
			}
		}
	}
}
