# Smart Notifications mod: TEMPORARY diagnostic for the law commitment
# alert (see TODO.md and
# common/alert_types/01_smart_notifications_alerts.txt's
# smart_notifications_law_commitment_alert). Added 2026-09-08 specifically
# so the alert's own logic (freshly rewritten to fix a scope bug never
# actually live-tested) can be confirmed from `debug.log` alone, without
# the user needing another round of clicking through the UI and waiting
# to see if an alert visually appears.
#
# `on_monthly_pulse_country` -- same real, already-confirmed-working hook
# the truce tracker uses (common/on_actions/02_smart_notifications_truce_tracker.txt).
# `is_player`-gated so this only ever logs for the player's own country.
#
# For every currently-flagged law (regardless of whether the alert's full
# condition is met), logs which of the two remaining gates
# (`can_be_enacted`, the success-vs-stall check) it currently passes
# or fails -- so if the alert never visibly appears, this line says
# exactly why, without needing new instrumentation added after the fact.
#
# REVISED 2026-09-08 per the user: the alert's own condition changed from
# a flat ">50% overall" threshold to "does the next checkpoint's Success
# chance beat its Stall chance" (matching the law list's own displayed
# breakdown) -- this probe's third check now mirrors that exact logic
# rather than the old threshold, so its output stays meaningful against
# whatever the alert itself currently does. See
# common/alert_types/01_smart_notifications_alerts.txt's own header
# comment for why this needs a 19-point threshold sweep instead of a
# single comparison.
#
# DELETE once the law commitment alert is confirmed working live.
#
# CORRECTED 2026-09-08 before ever running: `any_law` is trigger-only
# (confirmed: effects.log documents the effect-side iterator as a
# DIFFERENT keyword, `every_law` -- "Iterate through all laws in a
# country" -- while `any_law` only appears in triggers.log). Using
# `any_law` inside this file's `effect` block produced a real, confirmed
# "Unknown effect any_law" error live -- caught the same session as the
# alert's own `save_scope_as`-in-a-trigger mistake below, same general
# lesson: trigger-only and effect-only keywords that read almost
# identically are not interchangeable.
on_monthly_pulse_country = {
	on_actions = {
		smart_notifications_law_commitment_probe
	}
}

smart_notifications_law_commitment_probe = {
	effect = {
		if = {
			limit = { is_player = yes }
			every_law = {
				limit = { smart_notifications_law_matches_wanted_flag = yes }
				debug_log = "SNW_LAW_ALERT|probe|considering|law=[THIS.GetNameNoFormatting]"
				if = {
					limit = { can_be_enacted = yes }
					debug_log = "SNW_LAW_ALERT|probe|can_be_enacted=yes|law=[THIS.GetNameNoFormatting]"
					if = {
						limit = {
							THIS.owner = {
								OR = {
									AND = {
										enactment_chance_for_law = { target = prev.type  value > 0.05 }
										NOT = { stall_chance_for_law = { target = prev.type  value > 0.05 } }
									}
									AND = {
										enactment_chance_for_law = { target = prev.type  value > 0.1 }
										NOT = { stall_chance_for_law = { target = prev.type  value > 0.1 } }
									}
									AND = {
										enactment_chance_for_law = { target = prev.type  value > 0.15 }
										NOT = { stall_chance_for_law = { target = prev.type  value > 0.15 } }
									}
									AND = {
										enactment_chance_for_law = { target = prev.type  value > 0.2 }
										NOT = { stall_chance_for_law = { target = prev.type  value > 0.2 } }
									}
									AND = {
										enactment_chance_for_law = { target = prev.type  value > 0.25 }
										NOT = { stall_chance_for_law = { target = prev.type  value > 0.25 } }
									}
									AND = {
										enactment_chance_for_law = { target = prev.type  value > 0.3 }
										NOT = { stall_chance_for_law = { target = prev.type  value > 0.3 } }
									}
									AND = {
										enactment_chance_for_law = { target = prev.type  value > 0.35 }
										NOT = { stall_chance_for_law = { target = prev.type  value > 0.35 } }
									}
									AND = {
										enactment_chance_for_law = { target = prev.type  value > 0.4 }
										NOT = { stall_chance_for_law = { target = prev.type  value > 0.4 } }
									}
									AND = {
										enactment_chance_for_law = { target = prev.type  value > 0.45 }
										NOT = { stall_chance_for_law = { target = prev.type  value > 0.45 } }
									}
									AND = {
										enactment_chance_for_law = { target = prev.type  value > 0.5 }
										NOT = { stall_chance_for_law = { target = prev.type  value > 0.5 } }
									}
									AND = {
										enactment_chance_for_law = { target = prev.type  value > 0.55 }
										NOT = { stall_chance_for_law = { target = prev.type  value > 0.55 } }
									}
									AND = {
										enactment_chance_for_law = { target = prev.type  value > 0.6 }
										NOT = { stall_chance_for_law = { target = prev.type  value > 0.6 } }
									}
									AND = {
										enactment_chance_for_law = { target = prev.type  value > 0.65 }
										NOT = { stall_chance_for_law = { target = prev.type  value > 0.65 } }
									}
									AND = {
										enactment_chance_for_law = { target = prev.type  value > 0.7 }
										NOT = { stall_chance_for_law = { target = prev.type  value > 0.7 } }
									}
									AND = {
										enactment_chance_for_law = { target = prev.type  value > 0.75 }
										NOT = { stall_chance_for_law = { target = prev.type  value > 0.75 } }
									}
									AND = {
										enactment_chance_for_law = { target = prev.type  value > 0.8 }
										NOT = { stall_chance_for_law = { target = prev.type  value > 0.8 } }
									}
									AND = {
										enactment_chance_for_law = { target = prev.type  value > 0.85 }
										NOT = { stall_chance_for_law = { target = prev.type  value > 0.85 } }
									}
									AND = {
										enactment_chance_for_law = { target = prev.type  value > 0.9 }
										NOT = { stall_chance_for_law = { target = prev.type  value > 0.9 } }
									}
									AND = {
										enactment_chance_for_law = { target = prev.type  value > 0.95 }
										NOT = { stall_chance_for_law = { target = prev.type  value > 0.95 } }
									}
								}
							}
						}
						debug_log = "SNW_LAW_ALERT|probe|success_over_stall=yes|WOULD_ALERT"
					}
					else = {
						debug_log = "SNW_LAW_ALERT|probe|success_over_stall=no"
					}
				}
				else = {
					debug_log = "SNW_LAW_ALERT|probe|can_be_enacted=no|law=[THIS.GetNameNoFormatting]"
				}
			}
		}
	}
}
