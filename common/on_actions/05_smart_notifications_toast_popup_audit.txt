# TEMPORARY diagnostic audit -- DELETE once the user has reviewed a real
# session's SNW_TOAST_AUDIT counts and decided which of these are worth
# downgrading for unwatched countries (see TODO.md's "Toast/popup spam
# audit" item). Adds no player-visible behaviour: every block below only
# writes a debug_log line, mirroring the frequency-logger pattern already
# used in 01_smart_notifications_logger.txt.
#
# Purpose (2026-09-07, per the user): a real playtest is turning up
# repeated examples of toast notifications about countries the player has
# no reason to care about (Perak revolution, a decentralized rebel faction
# improving relations, etc.) -- but there's no single engine hook that
# fires for "every notification about to show as a toast/popup", so this
# was built by cross-referencing every message key CURRENTLY set to
# `notification_type = toast` or `popup` in 00_messages.txt against every
# vanilla on_action that posts it (found via a one-off script, not by
# reading the ~45k-line vanilla files by hand -- see CLAUDE.md's Token
# Budget rule).
#
# Coverage, out of 89 toast/popup keys (excluding this mod's own, already
# covered by SNW_FILTER):
#   - 45 keys are posted from a real on_action. Of those, ~17 were already
#     covered by 01_smart_notifications_logger.txt's existing SNW_LOG taps
#     (kept as-is, not duplicated here) -- the 28 on_actions below are the
#     genuinely new ones.
#   - 44 keys have NO moddable hook at all (fired from native engine code
#     directly -- same category already documented in TODO.md's Dev
#     Tooling section for country_attitude_changed/country_conscription/
#     invasion_*). Notably: all `power_bloc_*` notifications, all
#     `law_notification_*` keys, `election_results`, `country_revolution`,
#     `country_secession`, `invasion_started_against_us`,
#     `resource_discovered`. These can't be counted or filtered by script
#     at all -- manual observation is the only signal, same as the other
#     uncountable keys.
#
# To read the results after a session: grep debug.log for SNW_TOAST_AUDIT
# (this file) AND SNW_LOG (01_smart_notifications_logger.txt) together --
# both tags cover pieces of the full toast/popup picture, but SNW_LOG's
# existing hooks predate this audit and were curated for a different
# purpose (Phase 1 review), so some of what it counts (e.g. the diplo-play
# events) has since been replaced by this mod's own watched/quiet keys and
# no longer actually renders as a toast the way it did when that logger
# was built -- treat SNW_LOG counts for those specific keys as "how often
# the underlying event happens", not "how often the player saw a toast".
#
# No `is_player`/country-name scoping attempted anywhere below, on purpose
# -- root scope types vary too much across these 28 on_actions (Character,
# Country, Treaty, Formation, ...) to safely guess a shared accessor
# pattern, and CLAUDE.md's rule is to never guess script semantics. Plain,
# scope-free debug_log lines get the frequency count the user actually
# asked for without that risk. Where vanilla's own effect gates whether
# post_notification fires at all (on_acquired_technology, on_new_ruler,
# on_country_default's two loops), the same gate is mirrored here so the
# count isn't inflated relative to what the player actually saw.

on_acquired_technology = {
	on_actions = { smart_notifications_audit_acquired_technology }
}
smart_notifications_audit_acquired_technology = {
	effect = {
		if = {
			limit = { has_game_started = yes }
			debug_log = "SNW_TOAST_AUDIT|acquired_technology_notification"
		}
	}
}

on_new_ruler = {
	on_actions = { smart_notifications_audit_new_ruler }
}
smart_notifications_audit_new_ruler = {
	effect = {
		if = {
			limit = { owner = { has_gov_regency = no } }
			if = {
				limit = {
					owner = { NOT = { has_variable = hide_ruler_change_notification } }
					is_adult = yes
				}
				debug_log = "SNW_TOAST_AUDIT|new_ruler"
			}
		}
	}
}
